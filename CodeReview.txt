import SwiftUI

// MARK: - 液态玻璃设计系统
//
// 目标是 iOS 新版 Liquid Glass 的观感：半透明材质 + 背景色渗透 + 镜面高光描边 +
// 浮动柔和阴影 + 连续大圆角。用 Material 叠加自定义渐变描边实现，不依赖特定系统版本，
// 这样既能保持观感，又能在较低系统上正常安装运行。

enum GlassRadius {
    static let card: CGFloat = 26
    static let control: CGFloat = 18
    static let pill: CGFloat = 30
}

// MARK: - 组间歇配色
//
// 组间歇用绿色，和「训练」本身（accentColor 蓝）区分开：
// 蓝 = 我在练，绿 = 我在歇。只用于休息环的进度弧，
// 不动任何尺寸、线宽与动画参数，也不改训练页的蓝色强调色。

enum RestStyle {
    static let ring = Color.green
}

// MARK: - 训练配色
//
// 训练中（working）一律用蓝色强调色，组间歇（resting）一律用 RestStyle.ring 绿。
// 两阶段状态机：working --完成--> 记录一组 + resting --倒计时归零 / 跳过--> working …
// 「完成」只在 working 出现，「跳过休息」只在 resting 出现。

enum WorkStyle {
    static let ring = Color.accentColor
}

struct GlassBackgroundModifier: ViewModifier {
    @Environment(\.colorScheme) private var scheme
    var cornerRadius: CGFloat = GlassRadius.card
    var tint: Double = 0
    var shadow: Bool = true

    func body(content: Content) -> some View {
        content
            .background {
                ZStack {
                    RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                        .fill(.ultraThinMaterial)

                    // 纯色薄涂层：不再用渐变，避免页面出现脏色过渡
                    RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                        .fill(Color.white.opacity(scheme == .dark ? 0.06 : 0.22))

                    if tint > 0 {
                        RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                            .fill(Color.accentColor.opacity(tint))
                    }
                }
            }
            .overlay {
                // 单色细描边：玻璃边缘的实线，不做明暗渐变
                RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                    .strokeBorder(Color.white.opacity(scheme == .dark ? 0.14 : 0.55),
                                  lineWidth: 1)
            }
            .compositingGroup()
            .shadow(color: Color.black.opacity(shadow ? (scheme == .dark ? 0.45 : 0.10) : 0),
                    radius: shadow ? 22 : 0,
                    x: 0,
                    y: shadow ? 10 : 0)
    }
}

extension View {
    /// 液态玻璃卡片
    func glass(cornerRadius: CGFloat = GlassRadius.card, tint: Double = 0, shadow: Bool = true) -> some View {
        modifier(GlassBackgroundModifier(cornerRadius: cornerRadius, tint: tint, shadow: shadow))
    }

    /// 内层玻璃（嵌套在卡片里的次级容器，不投影）
    func glassInner(cornerRadius: CGFloat = GlassRadius.control) -> some View {
        modifier(GlassBackgroundModifier(cornerRadius: cornerRadius, tint: 0, shadow: false))
    }

    /// 页面级点击收起键盘
    func dismissKeyboardOnTap() -> some View {
        simultaneousGesture(
            TapGesture().onEnded {
                UIApplication.shared.sendAction(#selector(UIResponder.resignFirstResponder),
                                                to: nil, from: nil, for: nil)
            }
        )
    }
}

// MARK: - 卡片圆角裁剪（只在需要时才裁）

/// 圆角裁剪：**结构恒定，只有数值在变**。
///
/// 为什么不能直接用 `.clipShape`：
/// SwiftUI 里凡是不 ignore 安全区的容器（TabView、NavigationStack 底下的页面、
/// GeometryReader 也是这样）都是**按安全区布局**的。里面再放一个 `.ignoresSafeArea()` 的背景，
/// 背景是能画到状态栏和底部横条上的；可一旦在这条链路上套
/// `.clipShape(RoundedRectangle(cornerRadius: 0))`，裁剪边就是**安全区矩形**，
/// 连圆角为 0 都会硬切一刀，把铺出去的那部分削平 —— 露出来就是底下那层，
/// 表现为「一进 App 上下两条大黑边」「播放页上下两条大黑边」（4.6 → 4.8 的老 Bug）。
///
/// 为什么用 `mask` 而不是 `clipShape`：
/// mask 里的形状带了 `.ignoresSafeArea()`，圆角为 0 时它就是一块**覆盖整屏（含状态栏与
/// 底部横条）的矩形**，不沿安全区矩形切边，所以静止状态不可能切出黑边；
/// 需要圆角时把半径变大，遮罩自己变成圆角矩形，一切都还在整屏范围内。
///
/// 为什么不能写成 `if active { … } else { … }`：
/// 那是 `_ConditionalContent`，两个分支是不同的结构身份。`active` 一翻转，
/// SwiftUI 会把 `content` 整个销毁重建 —— 对根 TabView 来说就是：选中的 tab 回到第一页、
/// 各页 ScrollView 位置归零、NavigationStack 路径清空，而且过渡退化成淡入淡出而不是连续插值。
/// 现在 `body` 永远返回同一种结构（mask 一层），`active` 只影响 `effectiveRadius` 的数值，
/// 结构身份不变，状态不丢，圆角也能被外层 `.animation(…, value: player.isOpen)` 连续驱动。
struct CardClip: ViewModifier {
    /// false 时圆角按 0 处理（遮罩=整屏矩形，等于不裁）；true 时按 `radius` 裁圆角
    var active: Bool
    var radius: CGFloat

    private var effectiveRadius: CGFloat {
        (active && radius > 0.01) ? radius : 0
    }

    func body(content: Content) -> some View {
        content.mask {
            RoundedRectangle(cornerRadius: effectiveRadius, style: .continuous)
                .ignoresSafeArea()
        }
    }
}

import Foundation
import Combine
import AVFoundation
import AudioToolbox
import UserNotifications
import UIKit

/// 组间休息计时器。
/// 关键点：用「结束时刻的绝对时间戳」算剩余，不用累减计数 —— 这样 App 被切到后台、
/// 甚至 JS/RunLoop 被挂起，回来后剩余时间依然准确。后台到点由本地通知负责提醒。
final class RestTimer: NSObject, ObservableObject {
    static let shared = RestTimer()

    @Published private(set) var endDate: Date?
    @Published private(set) var total: Int = 120
    @Published private(set) var tick = Date()
    /// 当前这一组「开始做」的时刻。纯内存态，不落盘（重启后按 App 启动时刻算）。
    /// working 阶段的圈（本组用时）就靠它算 elapsed，不另开 Timer。
    @Published private(set) var workStart = Date()

    private var timer: Timer?
    private let notificationID = "reps.rest.finished"
    private var didRequestAuth = false

    var isRunning: Bool { endDate != nil }

    /// 两阶段状态机：working = !isResting，resting = isRunning。
    var isResting: Bool { isRunning }

    /// 开始做一组：把本组计时归零。skip() / finish() 末尾都会调用它，
    /// 所以「跳过休息」「倒计时自然归零」两条路径回到 working 时本组用时都从 0:00 起。
    func beginWork() {
        workStart = Date()
        tick = Date()
    }

    var remaining: Int {
        guard let end = endDate else { return 0 }
        return max(0, Int(ceil(end.timeIntervalSince(tick))))
    }

    var progress: Double {
        guard total > 0 else { return 0 }
        return min(1, max(0, 1 - Double(remaining) / Double(total)))
    }

    var remainingText: String {
        let r = remaining
        return String(format: "%d:%02d", r / 60, r % 60)
    }

    private override init() {
        super.init()
        NotificationCenter.default.addObserver(self, selector: #selector(enteredForeground),
                                               name: UIApplication.willEnterForegroundNotification, object: nil)
    }

    func requestNotificationAuthIfNeeded() {
        guard !didRequestAuth else { return }
        didRequestAuth = true
        UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { _, _ in }
    }

    // MARK: - 控制

    func start(seconds: Int) {
        requestNotificationAuthIfNeeded()
        total = max(5, seconds)
        endDate = Date().addingTimeInterval(TimeInterval(total))
        tick = Date()
        startTicking()
        scheduleNotification(after: TimeInterval(total))
        UIApplication.shared.isIdleTimerDisabled = true
    }

    func add(seconds: Int) {
        guard let end = endDate else { return }
        endDate = end.addingTimeInterval(TimeInterval(seconds))
        total = max(total, remaining)
        tick = Date()
        scheduleNotification(after: endDate?.timeIntervalSinceNow ?? 0)
    }

    func skip() {
        cancelNotification()
        endDate = nil
        stopTicking()
        UIApplication.shared.isIdleTimerDisabled = false
        // 跳过休息 = 直接进入下一组的 working 阶段
        beginWork()
    }

    private func finish() {
        // 先把排队中的通知撤掉：前台 tick 可能比通知触发早几十毫秒判定归零，
        // 若不取消，稍后 App 一进后台仍会弹一条「休息结束」，等于重复提醒。
        cancelNotification()
        endDate = nil
        stopTicking()
        UIApplication.shared.isIdleTimerDisabled = false
        // 前台提示：铃声 + 震动。后台提示走本地通知。
        AudioServicesPlaySystemSound(1005)
        AudioServicesPlaySystemSound(kSystemSoundID_Vibrate)
        let generator = UINotificationFeedbackGenerator()
        generator.notificationOccurred(.success)
        // 休息自然结束 = 下一组开始，本组计时归零
        beginWork()
    }

    // MARK: - 内部

    private func startTicking() {
        stopTicking()
        let t = Timer(timeInterval: 0.2, repeats: true) { [weak self] _ in
            guard let self = self else { return }
            self.tick = Date()
            if self.remaining <= 0 { self.finish() }
        }
        RunLoop.main.add(t, forMode: .common)
        timer = t
    }

    private func stopTicking() {
        timer?.invalidate()
        timer = nil
    }

    @objc private func enteredForeground() {
        tick = Date()
        if isRunning && remaining <= 0 { finish() }
    }

    private func scheduleNotification(after interval: TimeInterval) {
        cancelNotification()
        guard interval > 1 else { return }
        let content = UNMutableNotificationContent()
        content.title = "休息结束"
        content.body = "该上下一组了。"
        content.sound = .default
        let trigger = UNTimeIntervalNotificationTrigger(timeInterval: interval, repeats: false)
        let req = UNNotificationRequest(identifier: notificationID, content: content, trigger: trigger)
        UNUserNotificationCenter.current().add(req, withCompletionHandler: nil)
    }

    private func cancelNotification() {
        UNUserNotificationCenter.current().removePendingNotificationRequests(withIdentifiers: [notificationID])
        UNUserNotificationCenter.current().removeDeliveredNotifications(withIdentifiers: [notificationID])
    }
}

/// 点一下执行一次；按住 0.45 秒后开始连续触发，越按越快。
final class RepeatController: ObservableObject {
    var delay: Timer?
    var repeating: Timer?
    @Published var pressing = false

    func begin(_ action: @escaping () -> Void) {
        guard delay == nil && repeating == nil else { return }
        pressing = true
        action()
        delay = Timer.scheduledTimer(withTimeInterval: 0.45, repeats: false) { [weak self] _ in
            guard let self = self else { return }
            self.repeating = Timer.scheduledTimer(withTimeInterval: 0.08, repeats: true) { _ in
                action()
            }
        }
    }

    func end() {
        pressing = false
        delay?.invalidate(); delay = nil
        repeating?.invalidate(); repeating = nil
    }
}

struct RepeatButton: View {
    let systemImage: String
    var tint: Color = .primary
    let action: () -> Void

    @StateObject private var controller = RepeatController()

    var body: some View {
        Image(systemName: systemImage)
            .font(.system(size: 20, weight: .semibold))
            .foregroundColor(tint)
            .frame(width: 46, height: 46)
            .background(Color(UIColor.tertiarySystemFill))
            .clipShape(Circle())
            .scaleEffect(controller.pressing ? 0.92 : 1)
            .contentShape(Circle())
            .gesture(
                DragGesture(minimumDistance: 0)
                    .onChanged { _ in controller.begin(action) }
                    .onEnded { _ in controller.end() }
            )
    }
}

// MARK: - 折线图（自绘，不依赖 iOS 16 的 Charts）

struct Sparkline: View {
    let values: [Double]
    var lineColor: Color = .accentColor
    var height: CGFloat = 56

    var body: some View {
        GeometryReader { geo in
            let pts = points(in: geo.size)
            if pts.count >= 2 {
                ZStack {
                    Path { p in
                        p.move(to: pts[0])
                        for pt in pts.dropFirst() { p.addLine(to: pt) }
                    }
                    .stroke(lineColor, style: StrokeStyle(lineWidth: 2, lineCap: .round, lineJoin: .round))

                    if let last = pts.last {
                        Circle()
                            .fill(lineColor)
                            .frame(width: 7, height: 7)
                            .position(last)
                    }
                }
            } else {
                Text("数据还不够，再练一次就有趋势")
                    .font(.caption)
                    .foregroundColor(.secondary)
                    .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .leading)
            }
        }
        .frame(height: height)
    }

    private func points(in size: CGSize) -> [CGPoint] {
        guard values.count >= 2, size.width > 0 else { return [] }
        let mn = values.min() ?? 0
        let mx = values.max() ?? 1
        let range = (mx - mn) == 0 ? 1 : (mx - mn)
        let padX: CGFloat = 6
        let padY: CGFloat = 8
        return values.enumerated().map { i, v in
            let x = padX + CGFloat(i) / CGFloat(values.count - 1) * (size.width - padX * 2)
            let y = size.height - padY - CGFloat((v - mn) / range) * (size.height - padY * 2)
            return CGPoint(x: x, y: y)
        }
    }
}

// MARK: - 多指标双轴折线图

/// 一张图叠多条曲线，每条曲线用各自的 Y 轴刻度（量纲不同也能一起看）。
/// X 轴统一为「健身第几天」，所以体重、体脂、去脂体重、FFMI 可以同屏对比。
struct MetricChartView: View {
    let data: [Store.BodyMetricKey: [Store.ChartPoint]]

    private let leftPad: CGFloat = 46
    private let rightPad: CGFloat = 46
    private let topPad: CGFloat = 14
    private let bottomPad: CGFloat = 30

    private var activeKeys: [Store.BodyMetricKey] {
        Store.BodyMetricKey.allCases.filter { !(data[$0]?.isEmpty ?? true) }
    }

    private var allPoints: [Store.ChartPoint] {
        activeKeys.flatMap { data[$0] ?? [] }
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 10) {
            if allPoints.isEmpty {
                placeholder
            } else {
                chart
                legend
            }
        }
    }

    private var placeholder: some View {
        VStack(spacing: 8) {
            Image(systemName: "chart.xyaxis.line")
                .font(.system(size: 28))
                .foregroundColor(.secondary)
            Text("还没有身体数据")
                .font(.subheadline)
                .foregroundColor(.secondary)
            Text("在「我的」里填体重和体脂，每次记录都会变成一个点")
                .font(.caption)
                .foregroundColor(.secondary)
                .multilineTextAlignment(.center)
        }
        .frame(maxWidth: .infinity)
        .frame(height: 180)
    }

    private var chart: some View {
        // 先把所有依赖算成局部常量，避免 Canvas 的 escaping 闭包捕获 self
        let keys = activeKeys
        let points = allPoints
        let seriesData = data
        var rangeMap: [Store.BodyMetricKey: (low: Double, high: Double)] = [:]
        for k in keys {
            if let r = yRange(for: k) { rangeMap[k] = r }
        }
        let days = points.map { $0.day }
        let minDay = days.min() ?? 1
        let maxDay = days.max() ?? 2
        let daySpan = max(1, maxDay - minDay)
        let padL = leftPad
        let padR = rightPad
        let padT = topPad
        let padB = bottomPad
        let ranges = rangeMap

        return Canvas { context, size in
            let plot = CGRect(x: padL, y: padT,
                              width: max(1, size.width - padL - padR),
                              height: max(1, size.height - padT - padB))
            guard !keys.isEmpty else { return }

            let gridCount = 4
            for i in 0...gridCount {
                let ratio = CGFloat(i) / CGFloat(gridCount)
                let y = plot.maxY - ratio * plot.height

                var line = Path()
                line.move(to: CGPoint(x: plot.minX, y: y))
                line.addLine(to: CGPoint(x: plot.maxX, y: y))
                context.stroke(line,
                               with: .color(Color.primary.opacity(i == 0 ? 0.22 : 0.08)),
                               style: StrokeStyle(lineWidth: i == 0 ? 1.2 : 0.8,
                                                  dash: i == 0 ? [] : [3, 4]))

                // 左轴：第一条曲线
                if let key = keys.first, let range = ranges[key] {
                    let v = range.low + (range.high - range.low) * Double(ratio)
                    let label = Text(MetricChartView.formatAxis(v))
                        .font(.system(size: 9, weight: .medium, design: .rounded))
                        .foregroundColor(key.color.opacity(0.85))
                    context.draw(label, at: CGPoint(x: plot.minX - 8, y: y), anchor: .trailing)
                }

                // 右轴：第二条曲线
                if keys.count > 1, let key = keys.dropFirst().first, let range = ranges[key] {
                    let v = range.low + (range.high - range.low) * Double(ratio)
                    let label = Text(MetricChartView.formatAxis(v))
                        .font(.system(size: 9, weight: .medium, design: .rounded))
                        .foregroundColor(key.color.opacity(0.85))
                    context.draw(label, at: CGPoint(x: plot.maxX + 8, y: y), anchor: .leading)
                }
            }

            // 曲线
            for (index, key) in keys.enumerated() {
                guard let pts = seriesData[key], !pts.isEmpty, let range = ranges[key] else { continue }
                let mapped = pts.map { p -> CGPoint in
                    let x = plot.minX + CGFloat(Double(p.day - minDay) / Double(daySpan)) * plot.width
                    let y = plot.maxY - CGFloat((p.value - range.low) / max(0.0001, range.high - range.low)) * plot.height
                    return CGPoint(x: x, y: y)
                }

                if mapped.count >= 2 {
                    var path = Path()
                    path.move(to: mapped[0])
                    for p in mapped.dropFirst() { path.addLine(to: p) }

                    if index == 0 {
                        var area = path
                        area.addLine(to: CGPoint(x: mapped[mapped.count - 1].x, y: plot.maxY))
                        area.addLine(to: CGPoint(x: mapped[0].x, y: plot.maxY))
                        area.closeSubpath()
                        context.fill(area, with: .color(key.color.opacity(0.12)))
                    }

                    context.stroke(path,
                                   with: .color(key.color),
                                   style: StrokeStyle(lineWidth: index == 0 ? 2.4 : 1.8,
                                                      lineCap: .round, lineJoin: .round))
                }

                let showAll = mapped.count <= 12
                for (i, p) in mapped.enumerated() {
                    let isEdge = i == 0 || i == mapped.count - 1
                    guard showAll || isEdge else { continue }
                    let r: CGFloat = isEdge ? 3 : 2
                    let dot = Path(ellipseIn: CGRect(x: p.x - r, y: p.y - r, width: r * 2, height: r * 2))
                    context.fill(dot, with: .color(key.color))
                    context.stroke(dot, with: .color(Color(UIColor.systemBackground)), lineWidth: 1.2)
                }
            }

            let left = Text("第 \(minDay) 天")
                .font(.system(size: 9, weight: .medium, design: .rounded))
                .foregroundColor(.secondary)
            context.draw(left, at: CGPoint(x: plot.minX, y: plot.maxY + 16), anchor: .leading)

            if maxDay != minDay {
                let right = Text("第 \(maxDay) 天")
                    .font(.system(size: 9, weight: .medium, design: .rounded))
                    .foregroundColor(.secondary)
                context.draw(right, at: CGPoint(x: plot.maxX, y: plot.maxY + 16), anchor: .trailing)
            }
        }
        .frame(height: 210)
    }

    static func formatAxis(_ v: Double) -> String {
        if abs(v) >= 100 { return String(format: "%.0f", v) }
        return String(format: "%.1f", v)
    }

    private var legend: some View {
        HStack(spacing: 14) {
            ForEach(activeKeys) { key in
                HStack(spacing: 5) {
                    Circle().fill(key.color).frame(width: 7, height: 7)
                    Text(key.title + (key.unit.isEmpty ? "" : " (\(key.unit))"))
                        .font(.system(size: 10))
                        .foregroundColor(.secondary)
                }
            }
            Spacer()
        }
    }

    // MARK: - 计算

    private func yRange(for key: Store.BodyMetricKey) -> (low: Double, high: Double)? {
        let values = (data[key] ?? []).map { $0.value }
        guard let mn = values.min(), let mx = values.max() else { return nil }
        if mn == mx {
            let pad = max(0.5, abs(mn) * 0.05)
            return (mn - pad, mx + pad)
        }
        let pad = (mx - mn) * 0.12
        return (mn - pad, mx + pad)
    }

    private func formatAxis(_ v: Double, key: Store.BodyMetricKey) -> String {
        MetricChartView.formatAxis(v)
    }
}

    //

    // 支持拼音），这里只负责调用 + 顺手预热前几行的封面。

    private func runLibrarySearch() {
        let q = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard tab == .apple, am.isGranted, !q.isEmpty else {
            libraryHits = []
            albumHits = []
            didSearch = false
            return
        }
        didSearch = true
        libraryHits = am.search(q)
        albumHits = am.albums(from: libraryHits, matching: q)
        // 前几行的封面先排进后台，滚动时不至于一片灰
        ArtworkStore.shared.prefetch(libraryHits, side: 160, limit: 12)
    }

    // 队列仍是整串结果，persistentID 精确寻址，点哪首就是哪首。

    private var tileSide: CGFloat {
        let w = UIScreen.main.bounds.width - 40      // 页面左右各 20

    }

// MARK: - 全屏播放页
//
// 版式跟 Apple Music 对齐（左边距统一 24，所有元素共用一条中轴）：
//
//            ── 抓手 ──
//   ⌄          正在播放           ⋯
//        [      封面      ]      屏宽 76%，上限 322
//   歌名（22 粗，左对齐）
//   艺人（17，灰，左对齐）
//   ══════ 进度条 8pt ══════
//   0:12                  -3:20
//   ☰     ⏮     ▶     ⏭     🔁
//   🔈 ═════ 音量 ═════ 🔊
//   [ 训练状态卡 ]  ← 点开是训练面板
//
// 背景是灰色液态玻璃，整页固定深色，所以进度 / 音量条是白色。

struct NowPlayingView: View {
    @ObservedObject var hub = MusicHub.shared
    @ObservedObject var store = Store.shared
    @ObservedObject var rest = RestTimer.shared
    @ObservedObject var player = PlayerPresenter.shared

    @State private var dragOffset: CGFloat = 0
    /// 这一轮下拉有没有已经切过窗口配色（只在第一次 onChanged 里做一次）
    @State private var dismissing = false
    @State private var showQueue = false
    @State private var showTraining = false
    /// 拖动进度条时的临时秒数（nil = 没在拖，跟随播放器）
    @State private var scrub: Double?

    /// 全页统一左右留白：封面、标题、进度条、按钮、音量都对齐到这条线
    private let inset: CGFloat = 24

    var body: some View {
        GeometryReader { geo in
            let safe = SafeArea.insets
            // 除封面以外的固定高度（抓手 + 头部 + 标题 + 进度 + 控制 + 音量 + 训练卡）
            //
            // 4.9.2 由 448 加到 462，构成：标题区两行间距 +3（2→5）、
            // 训练卡文字间距 +4（4→6，两处间隙各 +2）、训练卡上下内边距 +4（12→14）、
            // 右侧三个小按钮间距 +2（8→9）、再留 1pt 余量。
            // 小屏预算（side = min(宽×0.82, 高 - 上下安全区 - 462, 340)）：
            //   iPhone SE3 375×667（安全区 20/0）→ 185pt，未触及 130 下限；
            //   iPhone 13 mini 375×812（50/34）→ 266pt；iPhone 16（59/34）→ 297pt。
            // 也就是说小屏上封面会变小一点，但不会被挤出屏幕（Spacer 还能再压）。
            let fixed: CGFloat = 462
            // 铺满整屏之后高度富余，封面按屏宽 82% 走，上限 340
            let side = max(130, min(geo.size.width * 0.82,
                                    geo.size.height - safe.top - safe.bottom - fixed,
                                    340))
            // 标题 / 进度 / 控制 / 音量都对齐到封面的左右边线（Apple Music 就是这个对齐）
            let pad = max(inset, (geo.size.width - side) / 2)

            VStack(spacing: 0) {
                grabber

                header

                Spacer(minLength: 10)

                cover(side: side)

                Spacer(minLength: 18)
                    .frame(height: 24)

                titleBlock(pad: pad)

                Spacer(minLength: 16)
                    .frame(height: 20)

                progressRow(pad: pad)

                Spacer(minLength: 12)
                    .frame(height: 18)

                controls(pad: pad)

                Spacer(minLength: 12)
                    .frame(height: 18)

                volume(pad: pad)

                Spacer(minLength: 10)

                trainingBar
                    .padding(.horizontal, 16)
            }
            // 忽略安全区再自己让回来：这样背景和圆角裁剪都覆盖到状态栏 / 底部横条，
            // 不会在上下各留一条黑边
            .padding(.top, safe.top)
            .padding(.bottom, max(safe.bottom, 10))
            .frame(width: geo.size.width, height: geo.size.height)
        }
        .ignoresSafeArea()
        .background { playerBackground }
        // 顺序必须是「背景 → CardClip → offset」：
        // 遮罩要盖在卡片本身上，再让整张卡片往下位移，圆角才会跟着走；
        // 反过来（先 offset 再 mask）遮罩会留在原地，拖动时圆角跑到屏幕角落去。
        .modifier(CardClip(active: dragOffset > 0.5, radius: min(34, dragOffset / 5)))
        .offset(y: max(0, dragOffset))
        .environment(\.colorScheme, .dark)
        .sheet(isPresented: $showQueue) {
            QueueSheet()
        }
        .sheet(isPresented: $showTraining) {
            TrainingSheet()
        }
        // 播放页铺满整屏，键盘不该再把它顶起来（那会露出底下那一层）
        .ignoresSafeArea(.keyboard, edges: .bottom)
        .onAppear {
            // 进来就把键盘收掉：否则会跟着播放页一起浮在下面，
            // 下拉时底下的搜索框还会被它顶着、甚至能被点到
            UIApplication.shared.sendAction(#selector(UIResponder.resignFirstResponder),
                                            to: nil, from: nil, for: nil)
        }
    }

    // MARK: 背景

    /// 灰色液态玻璃：灰底 + 半透明材质 + 中性柔光 + 薄白釉面。
    /// 只有灰度没有色相，所以不会发蓝发紫。
    private var playerBackground: some View {
        ZStack {
            Color(white: 0.13)

            Rectangle().fill(.ultraThinMaterial)

            Circle()
                .fill(Color.white.opacity(0.10))
                .frame(width: 420, height: 420)
                .blur(radius: 130)
                .offset(x: -110, y: -260)

            Circle()
                .fill(Color.white.opacity(0.05))
                .frame(width: 460, height: 460)
                .blur(radius: 150)
                .offset(x: 130, y: 300)

            Color.white.opacity(0.03)
        }
        .environment(\.colorScheme, .dark)
        .ignoresSafeArea()
        .allowsHitTesting(false)
    }

    // MARK: 抓手

    private var grabber: some View {
        Capsule()
            .fill(Color.secondary.opacity(0.30))
            .frame(width: 36, height: 5)
            .padding(.top, 6)
            .padding(.bottom, 4)
            .frame(maxWidth: .infinity)
            .contentShape(Rectangle())
            .gesture(dismissDrag)
    }

    // MARK: 顶部

    private var header: some View {
        HStack(spacing: 0) {
            Button {
                player.close()
            } label: {
                Image(systemName: "chevron.down")
                    .font(.system(size: 17, weight: .semibold))
                    .foregroundColor(.primary)
                    .frame(width: 44, height: 44)
                    .contentShape(Rectangle())
            }
            .buttonStyle(.plain)

            Spacer(minLength: 0)

            Text("正在播放")
                .font(.system(size: 11, weight: .semibold))
                .foregroundColor(.secondary)
                .textCase(.uppercase)
                .tracking(0.6)

            Spacer(minLength: 0)

            Menu {
                Button {
                    hub.cycleRepeatMode()
                } label: {
                    Label(hub.repeatMode.label, systemImage: hub.repeatMode.symbol)
                }
                Button {
                    showQueue = true
                } label: {
                    Label("播放列表", systemImage: "list.bullet")
                }
                Button {
                    showTraining = true
                } label: {
                    Label("打开训练面板", systemImage: "figure.strengthtraining.traditional")
                }
                if let u = hub.current?.externalURL {
                    Button {
                        UIApplication.shared.open(u)
                    } label: {
                        Label("在 Apple Music 打开", systemImage: "arrow.up.forward.app")
                    }
                }
            } label: {
                Image(systemName: "ellipsis.circle")
                    .font(.system(size: 19, weight: .regular))
                    .foregroundColor(.primary)
                    .frame(width: 44, height: 44)
                    .contentShape(Rectangle())
            }
        }
        .padding(.horizontal, 8)
        .contentShape(Rectangle())
        .gesture(dismissDrag)
    }

    /// 往下拖收回：底下的菜单一直可见（并被一起带着放大 / 变亮），所以不会出现黑边；
    /// 拖过阈值就关，没拖够就弹回原位。
    ///
    /// ★ 坐标空间必须是 `.global`，这是 4.9.3 修「下拉时持续抖动 / 闪烁」的关键：
    /// 这个手势挂在抓手 / 头部 / 封面 / 标题区上，而这几个视图都在**被 `.offset(y: dragOffset)`
    /// 移动的卡片内部**。用默认的 `.local`（视图自身坐标）时，手指往下一动 → 视图跟着往下移 →
    /// 手指在这个局部坐标里的 y 被抵消 → `translation.height` 变小 → `dragOffset` 变小 →
    /// 卡片弹回去 → 下一帧 translation 又变大 …… 形成一个正反馈回路，表现为持续抖动，
    /// 而且 `player.pull` 也跟着来回跳，底下 TabView 的缩放 / 压暗 / 圆角同步抖。
    /// `.global` 不随卡片位移变化，拿到的才是手指的真实位移，dragOffset 单调跟随。
    /// （4.9.4 又验证了一次：抖动是坐标空间的问题，和窗口配色切换是两件事，不要混在一起。）
    private var dismissDrag: some Gesture {
        DragGesture(minimumDistance: 8, coordinateSpace: .global)
            .onChanged { v in
                let h = max(0, v.translation.height)
                // 拖动第一帧：位移只有几 pt、底下几乎还没露出来，就在这一帧把窗口切回浅色。
                // 这样整个下拉过程中露出来的页面，从头到尾都是最终那一版浅色，不会跳变。
                if h > 0 && !dismissing {
                    dismissing = true
                    player.setDragging(true)
                    player.beginDismiss()
                }
                dragOffset = h
                // 0…1：拉到 260pt 就算到底，下层同步动
                player.pull = min(1, h / 260)
            }
            .onEnded { v in
                dismissing = false
                player.setDragging(false)
                let far = v.translation.height > 110 || v.predictedEndTranslation.height > 260
                if far {
                    // 配色在 beginDismiss 里已经切好，这里直接走 close() 的下滑分支
                    player.close()
                } else {
                    withAnimation(.spring(response: 0.34, dampingFraction: 0.86)) {
                        dragOffset = 0
                        player.pull = 0
                    }
                    // 弹回原位后再把窗口切回深色（0.36s 后，弹回动画 0.34s）；
                    // 若这期间用户又开始拖，回调里有 isDragging 守卫，不会被拉回深色。
                    player.cancelDismiss()
                }
            }
    }

    private func cover(side: CGFloat) -> some View {
        // 封面尺寸直接跟 side 走，**不加任何隐式动画**。
        // 以前这里挂过 .animation(.easeOut(duration: 0.25), value: side)：side 在拖动
        // （安全区回退 / 布局重算）时会变，这个动画让封面相对卡片滞后 0.25s，
        // 拖动过程中就会出现「封面和卡片错位」的撕裂帧。
        TrackArt(title: hub.current?.title ?? "REPS",
                 mediaID: hub.current?.mediaID,
                 size: side,
                 corner: 8)
            .shadow(color: .black.opacity(0.30), radius: 24, y: 16)
            .gesture(dismissDrag)
    }

        VStack(alignment: .leading, spacing: 5) {
            Text(hub.current?.title ?? "没有正在播放的音乐")
                .font(.system(size: 24, weight: .semibold))
                .lineSpacing(2)
                .foregroundColor(.primary)
                .lineLimit(1)
                .minimumScaleFactor(0.6)

            Text(subtitleText)
                .font(.system(size: 18, weight: .regular))
                .lineSpacing(2)
                .foregroundColor(.secondary)
                .lineLimit(1)
                .minimumScaleFactor(0.7)
        }
        .frame(maxWidth: .infinity, alignment: .leading)
        .padding(.horizontal, pad)
        .contentShape(Rectangle())
        .gesture(dismissDrag)
    }

    private var subtitleText: String {
        guard let c = hub.current else { return "" }
        return c.artist.isEmpty ? c.subtitle : c.artist
    }

    // MARK: 进度

    private func progressRow(pad: CGFloat) -> some View {
        VStack(spacing: 6) {
            GlassBar(progress: progressRatio,
                     onScrub: { v in scrub = v * max(1, hub.duration) },
                     onCommit: { v in
                         hub.seek(to: v * max(1, hub.duration))
                         scrub = nil
                     },
                     trackHeight: 8)

            HStack(spacing: 0) {
                Text(MusicHub.formatTime(elapsed))
                Spacer(minLength: 8)
                Text("-" + MusicHub.formatTime(max(0, hub.duration - elapsed)))
            }
            .font(.system(size: 12, weight: .medium))
            .monospacedDigit()
            .foregroundColor(.secondary)
        }
        .padding(.horizontal, pad)
    }

    private var progressRatio: Double {
        let d = max(1, hub.duration)
        if let s = scrub { return min(max(s / d, 0), 1) }
        return min(max(hub.currentTime / d, 0), 1)
    }

    private var elapsed: Double {
        scrub ?? hub.currentTime
    }

    // MARK: 播放控制
    //
    // Apple Music 是「随机 / 上一首 / 播放 / 下一首 / 循环」，第一个位置这里换成播放列表。

    private func controls(pad: CGFloat) -> some View {
        HStack(spacing: 0) {
            Button {
                showQueue = true
            } label: {
                Image(systemName: "list.bullet")
                    .font(.system(size: 21, weight: .regular))
                    .foregroundColor(.primary)
                    .frame(width: 50, height: 50)
                    .contentShape(Rectangle())
            }
            .buttonStyle(.plain)

            Spacer(minLength: 0)

            Button {
                hub.previous()
            } label: {
                Image(systemName: "backward.fill")

// MARK: - 玻璃进度条 / 音量条
//
// 照 Apple Music 的观感做：8pt 厚的胶囊条，浅灰半透明轨道 +
// 已播放段是白色（深色模式）/ 近黑（浅色模式），0.5pt 高光描边 + 轻微外发光，
// 看起来像一根磨过的玻璃条，而不是系统 Slider 那种塑料件。
// 没有圆点把手（按住任意位置拖，命中区 26pt），拖动时按下位置跟手。

struct GlassBar: View {
    var progress: Double
    var onScrub: (Double) -> Void = { _ in }
    var onCommit: (Double) -> Void = { _ in }
    var trackHeight: CGFloat = 8
    var enabled: Bool = true

    @Environment(\.colorScheme) private var scheme
    @State private var dragging = false
    @State private var local: Double = 0

    var body: some View {
        GeometryReader { geo in
            let w = max(1, geo.size.width)
            let p = CGFloat(min(max(dragging ? local : progress, 0), 1))

            ZStack(alignment: .leading) {
                // 轨道：玻璃槽
                Capsule(style: .continuous)
                    .fill(Color.primary.opacity(scheme == .dark ? 0.22 : 0.12))
                    .overlay(
                        Capsule(style: .continuous)
                            .strokeBorder(Color.white.opacity(scheme == .dark ? 0.12 : 0.45),
                                          lineWidth: 0.5)
                    )
                    .frame(height: trackHeight)

                // 已播放段：白 / 深灰玻璃
                Capsule(style: .continuous)
                    .fill(fillColor)
                    .overlay(
                        Capsule(style: .continuous)
                            .strokeBorder(Color.white.opacity(scheme == .dark ? 0.65 : 0.30),
                                          lineWidth: 0.5)
                    )
                    .frame(height: trackHeight)
                    .mask(alignment: .leading) {
                        Rectangle().frame(width: w * p)
                    }
                    .shadow(color: fillColor.opacity(scheme == .dark ? 0.45 : 0.35), radius: 4, y: 0)
            }
            .frame(width: w, height: max(trackHeight, 26))
            .contentShape(Rectangle())
            .gesture(
                DragGesture(minimumDistance: 0)
                    .onChanged { v in
                        guard enabled else { return }
                        if !dragging {
                            dragging = true
                            UISelectionFeedbackGenerator().selectionChanged()
                        }
                        let t = min(max(v.location.x / w, 0), 1)
                        local = t
                        onScrub(t)
                    }
                    .onEnded { v in
                        guard enabled else { return }
                        let t = min(max(v.location.x / w, 0), 1)
                        local = t
                        dragging = false
                        onCommit(t)
                    }
            )
        }
        .frame(height: max(trackHeight, 26))
        .opacity(enabled ? 1 : 0.4)
    }

    private var fillColor: Color {
        scheme == .dark
            ? Color.white.opacity(0.97)
            : Color(white: 0.12).opacity(0.86)
    }
}

struct VolumeBar: View {
    var trackHeight: CGFloat = 8

    @State private var level: Double = SystemVolume.shared.level
    @State private var dragging = false
    @State private var local: Double = 0

    private let ticker = Timer.publish(every: 0.25, on: .main, in: .common).autoconnect()

    var body: some View {
        GlassBar(progress: dragging ? local : level,
                 onScrub: { v in
                     dragging = true
                     local = v
                     SystemVolume.shared.set(v)
                 },
                 onCommit: { v in
                     local = v
                     level = v
                     dragging = false
                     SystemVolume.shared.set(v)
                 },
                 trackHeight: trackHeight)
            .onReceive(ticker) { _ in
                guard !dragging else { return }
                let v = SystemVolume.shared.level
                if abs(v - level) > 0.005 { level = v }
            }
            .onAppear { level = SystemVolume.shared.level }
    }
}

/// 系统音量读写。
///
/// iOS 不开放改音量的 API，只能借 MPVolumeView 内部那个 UISlider：
/// 把它挂到一个看不见的容器里，读它的 value / 给它 setValue，
/// 就能做到和侧边键完全同步的真实音量控制。
final class SystemVolume {
    static let shared = SystemVolume()

    private let host = MPVolumeView(frame: CGRect(x: -4, y: -4, width: 1, height: 1))

    private var slider: UISlider? {
        host.subviews.compactMap { $0 as? UISlider }.first
    }

    private init() {
        host.showsRouteButton = false
        host.alpha = 0.01
        host.isUserInteractionEnabled = false
    }

    /// MPVolumeView 的 UISlider 是懒创建的，先进窗口再取一次。
    private func install() {
        if host.superview == nil {
            guard let scene = UIApplication.shared.connectedScenes
                    .compactMap({ $0 as? UIWindowScene })
                    .first(where: { $0.activationState == .foregroundActive })
                    ?? UIApplication.shared.connectedScenes.compactMap({ $0 as? UIWindowScene }).first,
                  let window = scene.windows.first(where: { $0.isKeyWindow }) ?? scene.windows.first
            else { return }
            window.addSubview(host)
            host.setNeedsLayout()
            host.layoutIfNeeded()
        }
        if slider == nil {
            host.setNeedsLayout()
            host.layoutIfNeeded()
        }
    }

    var level: Double {
        install()
        if let s = slider { return Double(s.value) }
        return Double(AVAudioSession.sharedInstance().outputVolume)
    }

    func set(_ v: Double) {
        install()
        let target = Float(min(max(v, 0), 1))
        if let s = slider {
            s.setValue(target, animated: false)
        }
    }
}

/// 本地文件没有内嵌封面时，用歌名挑一个纯色占位（不用渐变）。
struct AlbumArt: View {
    let title: String
    var size: CGFloat

    var body: some View {
        ZStack {
            RoundedRectangle(cornerRadius: size * 0.08, style: .continuous)
                .fill(Self.baseColor(for: title))
            Image(systemName: "music.note")
                .font(.system(size: size * 0.2, weight: .light))
                .foregroundColor(.white.opacity(0.92))
        }
        .frame(width: size, height: size)
    }

    static func baseColor(for title: String) -> Color {
        let colors: [Color] = [
            Color(red: 0.32, green: 0.42, blue: 0.92),
            Color(red: 0.92, green: 0.44, blue: 0.38),
            Color(red: 0.16, green: 0.66, blue: 0.56),
            Color(red: 0.80, green: 0.36, blue: 0.58),
            Color(red: 0.24, green: 0.60, blue: 0.82)
        ]
        let seed = title.unicodeScalars.reduce(0) { $0 &+ Int($1.value) }
        return colors[abs(seed) % colors.count]
    }
}

import SwiftUI
import UIKit

// MARK: - 安全区
//
// 播放页要铺满整屏（含状态栏 / 底部横条区域），所以它自己忽略安全区，
// 再按真实安全区把内容往里让 —— 否则 `.clipShape` 会按安全区内的 frame 裁剪，
// 上下各留一条黑边（4.7 的「重大 Bug」就是这么来的）。

enum SafeArea {
    /// 取不到窗口时的回退值：按灵动岛机型（top 59 / bottom 34）。
    /// 绝不能用 0 —— 0 会让播放页内容顶到状态栏和底部横条下面。
    private static let fallback = UIEdgeInsets(top: 59, left: 0, bottom: 34, right: 0)

    static var insets: UIEdgeInsets {
        let scenes = UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }
        // 优先前台活跃的场景，没有再退回任意一个
        let scene = scenes.first(where: { $0.activationState == .foregroundActive }) ?? scenes.first
        let win = scene?.windows.first(where: { $0.isKeyWindow }) ?? scene?.windows.first
        // 首帧可能还没布局：top == 0 视为「拿不到」，回退到灵动岛基准值
        guard let insets = win?.safeAreaInsets, insets.top > 0 else { return fallback }
        return insets
    }
}

// MARK: - 播放页的呈现方式
//
// 以前用 fullScreenCover：那一层是完全独立的窗口层，往下拖时下面什么都没有，
// 露出来就是大片黑色。改成在根视图里叠一张自己管理的卡片：
// 打开时底下的 TabView 稍微缩小并压暗，播放页从底部推上来，
// 往下拖的时候底下的菜单一直看得见 —— 就是 iOS 那种「回到上一级」的观感。

final class PlayerPresenter: ObservableObject {
    static let shared = PlayerPresenter()
    @Published var isOpen = false
    /// 下拉进度：0 = 完全打开，1 = 已经拉到底（松手就关）。
    /// 下层菜单靠它跟着渐进放大 / 变亮 / 变圆角，而不是等关闭动画才动一下。
    @Published var pull: CGFloat = 0
    /// 播放页开着时把整个 App 切到深色（状态栏文字才会转白）。
    /// 不能随手写 `.preferredColorScheme(.dark)`：那会在播放页还在滑入、菜单露在外面的
    /// 那 0.4 秒里把整个窗口换成深色，看起来就是一整块颜色闪一下。
    /// 所以改成「完全盖住之后再切、开始下滑之前就切回来」——两次切换都发生在被遮住的时候。
    @Published var forceDark = false

    private init() {}

    func open() {
        guard !isOpen else { return }
        // 先收键盘：键盘弹着的时候迷你播放条被顶到键盘上方，点它打开播放页，
        // 键盘不会自己消失，会跟着播放页一起浮在下面（还能点到搜索框）。
        UIApplication.shared.sendAction(#selector(UIResponder.resignFirstResponder),
                                        to: nil, from: nil, for: nil)
        pull = 0
        withAnimation(.spring(response: 0.46, dampingFraction: 0.90)) {
            isOpen = true
        }
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.52) { [weak self] in
            // 0.52s 内用户已经把它关掉了：不要再把窗口切成深色，
            // 否则下一次打开会在「已经盖满」之前就变成深色，露出来那一下就是闪。
            // 也不要在「用户已经在下拉」的时候切：beginDismiss 刚把窗口切成浅色，
            // 这里再切回深色，等于在拖动中途翻色 —— 正是要修的那道跳变。
            // （这种极端情况由 cancelDismiss 的延迟回调兜底：松手弹回后 0.36s 会补回深色。）
            guard let self = self, self.isOpen, !self.isDragging else { return }
            self.forceDark = true
        }
    }

    // MARK: 配色切换的三个时机

    /// 往下拖的**第一帧**：位移还只有 8pt 左右，底下几乎还没露出来，
    /// 在这一帧把窗口切回浅色谁也看不见。
    ///
    /// 为什么必须这么早（4.9.4 的结论）：下拉期间底下露出的是哪一版 TabView，
    /// 完全取决于 forceDark。如果拖动全程保持深色（4.9.3 的做法），露出来的就是深色版页面，
    /// 等松手后再切浅 —— 那一整页当场由深变亮，就是用户看到的「深色/亮色跳变」。
    /// 所以要在一开始就把配色定成最终的样子。
    ///
    /// 用关掉动画的 Transaction：preferredColorScheme 的变化一旦被动画带，
    /// 会变成一段 0.4s 的渐变，叠在手势上更花。
    func beginDismiss() {
        guard forceDark else { return }
        var t = Transaction()
        t.disablesAnimations = true
        withTransaction(t) { forceDark = false }
    }

    /// 是否正在拖动。普通属性、不是 @Published —— 它只给 cancelDismiss 的延迟回调做守卫，
    /// 不需要触发任何视图刷新。
    private(set) var isDragging = false

    func setDragging(_ v: Bool) {
        isDragging = v
    }

    /// 没拖够、弹回原位：等弹回动画（response 0.34）走完再把窗口切回深色。
    /// 回调里要复查三件事：
    ///   · isOpen —— 这 0.36s 内可能已经被关掉；
    ///   · pull == 0 —— 确实已经回位；
    ///   · !isDragging —— 用户可能又开始拖了，那配色归 beginDismiss 管，不能被这里拉回深色。
    func cancelDismiss() {
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.36) { [weak self] in
            guard let self = self, self.isOpen, self.pull == 0, !self.isDragging else { return }
            self.forceDark = true
        }
    }

    func close() {
        guard isOpen else { return }
        // ★ 这里**不能**立刻 pull = 0：
        // pull 同时驱动下层的大小（0.97 + 0.03 * pull）和圆角（22 - 12 * pull）。
        // 手势拉到底时 pull 已经接近 1，如果关闭的一瞬间把它归零，
        // 下层会从「几乎原样」猛地缩回 0.97 的小卡片 —— 就是滑到底松手时的「跳一下」。
        // 所以先让 isOpen=false 的动画从当前 pull 继续插值，动画结束（0.6s）后再悄悄复位。
        // 先把窗口切回浅色。此刻播放页还盖满全屏，切了看不见。
        // 然后让配色先单独渲染一帧（0.05s），再启动下滑动画 ——
        // 这样播放页滑出的整个过程里，底下露出来的都是已经切好的浅色页面。
        if forceDark {
            var t = Transaction()
            t.disablesAnimations = true
            withTransaction(t) { forceDark = false }
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.05) { [weak self] in
                self?.startDismissAnimation()
            }
        } else {
            // 拖动开始时已经切过浅色了，不用再等
            startDismissAnimation()
        }
        // 关闭动画（spring response 0.40）走完之后再复位 pull。
        // 此时 isOpen 已经是 false，CardClip 不再生效、缩放也是 1，
        // 所以这次复位在视觉上没有作用 —— 纯粹为下一次打开准备干净的起点。
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.6) { [weak self] in
            guard let self = self, !self.isOpen else { return }
            self.pull = 0
        }
    }

    private func startDismissAnimation() {
        withAnimation(.spring(response: 0.40, dampingFraction: 0.92)) {
            isOpen = false
        }
    }
}

// MARK: - 训练面板
//
// 从播放页的训练卡点进来。内容就是训练页那套（同一个 Store / RestTimer），

struct TrainingSheet: View {

                        .frame(width: 44, height: 32)

                .buttonStyle(ScaleButtonStyle())
            }
        }
        .padding(16)
        .glass(cornerRadius: 20, shadow: false)
    }

    @ViewBuilder
    private var phaseRing: some View {
        if rest.isResting {
            MiniRestRing(progress: rest.progress, text: rest.remainingText,
                         size: 58, color: RestStyle.ring)
                .transition(.opacity)
        } else {
            TimelineView(.periodic(from: .now, by: 1)) { ctx in
                let elapsed = max(0, ctx.date.timeIntervalSince(rest.workStart))
                MiniRestRing(progress: elapsed.truncatingRemainder(dividingBy: 60) / 60,
                             text: clockText(elapsed),
                             size: 58,
                             color: WorkStyle.ring)
            }
            .transition(.opacity)
        }
    }

    private var phaseTitle: String {
        guard store.current != nil else { return "还没开始训练" }
        return rest.isResting ? "组间休息" : "第 \(setCount + 1) 组 · 训练中"
    }

    private var phaseSubtitle: String {
        guard store.current != nil else {
            return "调好重量和次数，点「完成一组」开始"
        }
        let target = "目标 \(formatWeight(store.lastWeight)) kg × \(store.lastReps)"
        return rest.isResting
            ? "下一组 \(formatWeight(store.lastWeight)) kg × \(store.lastReps)"
            : target
    }

    private var setCount: Int {
        store.current == nil ? 0 : store.currentExercise.sets.count
    }

    /// 主按钮：working 蓝「完成一组」/ resting 绿「跳过休息」，同一时刻只有一个
    @ViewBuilder
    private var primaryButton: some View {
        if rest.isResting {
            Button {
                rest.skip()
            } label: {
                Text("跳过休息")
            }
            .buttonStyle(GlassButtonStyle(prominent: true, tint: RestStyle.ring))
            .transition(.opacity)
        } else {
            Button {
                completeSet()
            } label: {
                Text("完成一组")
            }
            .buttonStyle(GlassButtonStyle(prominent: true))
            .transition(.opacity)
        }
    }

    // MARK: 组列表

    private var setListCard: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("本次记录")
                .font(.system(size: 12, weight: .semibold))
                .foregroundColor(.secondary)

            let sets = store.currentExercise.sets
            if sets.isEmpty {
                Text("还没有记录，调好重量和次数点「完成一组」")
                    .font(.system(size: 12))
                    .lineSpacing(3)
                    .foregroundColor(.secondary)
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding(.vertical, 6)
            } else {
                ForEach(Array(sets.enumerated()), id: \.element.id) { i, s in
                    HStack(spacing: 10) {
                        Text("\(i + 1)")
                            .font(.system(size: 11, weight: .semibold, design: .rounded))
                            .foregroundColor(.secondary)
                            .frame(width: 18, alignment: .trailing)
                        Text("\(formatWeight(s.w)) kg")
                            .font(.system(size: 14, weight: .medium, design: .rounded))
                        Text("× \(s.r)")
                            .font(.system(size: 13, design: .rounded))
                            .foregroundColor(.secondary)
                        Spacer(minLength: 4)
                        Button {
                            store.deleteSet(s.id)
                        } label: {
                            Image(systemName: "minus.circle")
                                .font(.system(size: 15))
                                .foregroundColor(.secondary)
                        }
                        .buttonStyle(.plain)
                    }
                    .padding(.vertical, 6)
                }
            }
        }
        .frame(maxWidth: .infinity, alignment: .leading)
        .padding(18)
        .glass(cornerRadius: 20, shadow: false)
    }

    private func completeSet() {
        // 「完成」只在 working 阶段有效：休息中不能记组
        guard !rest.isResting else { return }
        let hadSession = store.current != nil
        _ = store.ensureSession()
        if !hadSession { rest.beginWork() }
        store.logSet(weight: store.lastWeight, reps: store.lastReps)
        UIImpactFeedbackGenerator(style: .medium).impactOccurred()
        rest.start(seconds: store.restSeconds)
    }

    /// 秒数 → m:ss（本组用时）
    private func clockText(_ t: TimeInterval) -> String {
        let s = max(0, Int(t))
        return String(format: "%d:%02d", s / 60, s % 60)
    }

    private func formatWeight(_ w: Double) -> String {
        w.truncatingRemainder(dividingBy: 1) == 0
            ? String(format: "%.0f", w)
            : String(format: "%.1f", w)
    }
}

/// 底部栏用系统原生的 TabView（UIKit 的 UITabBar）：
/// 点按回弹、长按、滚动到底部再点当前页回顶、无障碍朗读、
/// 安全区处理和玻璃材质全部交给系统，新版 iOS 上自动是新外观。
struct RootTabView: View {
    @Environment(\.scenePhase) private var scenePhase
    @ObservedObject var player = PlayerPresenter.shared
    @State private var selection = 0

    var body: some View {
        ZStack {
            // 底层只有一层：**页面同色 + 铺满安全区**。
            //
            // 4.9.6 之前这里还叠过一层 `Color(white: 0.13)`（播放页深灰）当卡片堆叠的底，
            // 但那层没有 ignoresSafeArea：缩小后的页面四周就露出一圈深灰，
            // 上下安全区又因为它是按安全区布局的而露出浅灰 —— 灰带 + 黑圈，
            // 和状态栏 / 电池区完全割裂，下拉时特别明显。现在整层删掉，
            // AppBackground() 自己带 ignoresSafeArea，露出来的永远是页面同色，四周接得住。
            AppBackground()

            TabView(selection: $selection) {
                TrainView()
                    .tabItem {
                        Label("训练", systemImage: "figure.strengthtraining.traditional")
                    }
                    .tag(0)

                MusicView()
                    .tabItem {
                        Label("音乐", systemImage: "music.note")
                    }
                    .tag(1)

                GrowthView()
                    .tabItem {
                        Label("成长", systemImage: "chart.xyaxis.line")
                    }
                    .tag(2)

                ProfileView()
                    .tabItem {
                        Label("我的", systemImage: "person.crop.circle")
                    }
                    .tag(3)
            }
            .tint(.accentColor)
            // ★ 4.9 修「一进 App 上下两条大黑边」的真正原因：
            // 以前这里无条件套着 `.clipShape(RoundedRectangle(...))`。TabView 是按安全区
            // 布局的，一旦套上裁剪，即使圆角是 0，也会沿「安全区矩形」硬切一刀 ——
            // 页面延伸到状态栏 / 底部横条上的那部分（导航栏底、标签栏底）被切掉，
            // 露出的就是底下那层，于是上下各留一条边。
            // 现在只在播放页打开（卡片堆叠、需要圆角）时才裁，静止状态完全不裁。
            .modifier(CardClip(active: player.isOpen, radius: 22 - 12 * player.pull))
            // 跟着下拉进度实时变化：拉得越多，下层越接近原始大小、越亮 —— 就是「渐渐回到上一级」
            // 缩放幅度从 6% 收到 3%（0.97 + 0.03 * pull）：原来的 0.94 在拖到一半时
            // 把整页压得太小，内容挤成一坨。现在只是「轻微退后一级」，文字不再明显变形。
            .scaleEffect(player.isOpen ? 0.97 + 0.03 * player.pull : 1, anchor: .center)
            .overlay {
                // 压暗遮罩抽成了独立小视图，见文件底部的 DimOverlay：
                // pull 每帧都在变，遮罩只依赖 pull，没必要让根视图重新构造它。
                DimOverlay()
            }
            // 播放页盖在上面时，底下这层不接受点击：否则键盘弹着的时候，
            // 下拉露出来的搜索框还能被戳到。用 allowsHitTesting + accessibilityHidden，
            // 不额外加 .disabled —— .disabled 会把底下所有按钮画成灰态，
            // 而下拉过程中这层是看得见的，按钮会一个个变灰（视觉副作用明显）。
            .allowsHitTesting(!player.isOpen)
            .accessibilityHidden(player.isOpen)
            .animation(.spring(response: 0.46, dampingFraction: 0.90), value: player.isOpen)

            if player.isOpen {
                NowPlayingView()
                    .zIndex(10)
                    // 移出时只位移、不淡出：淡出会让播放页在半透明的那几帧里
                    // 透出底下那层（深浅色交接处就会闪一下）。滑出时点一下就能盖住。
                    .transition(.asymmetric(
                        insertion: .move(edge: .bottom).combined(with: .opacity),
                        removal: .move(edge: .bottom)
                    ))
            }
        }
        // 状态栏文字转白：只在播放页已经盖满全屏之后才切（切早了会整块闪）
        .preferredColorScheme(player.forceDark ? ColorScheme.dark : nil)
        .onChange(of: scenePhase) { phase in
            if phase == .background {
                Store.shared.writeNow()
            }
        }
    }
}

// `CardClip`（只在播放页打开时才给 TabView 裁圆角）定义在 Theme.swift，
// 播放页 NowPlayingView 用的是同一个修饰符。

// MARK: - 压暗遮罩
//
// 单独抽成一个只观察 player 的小视图：`pull` 是 @Published，拖动时每帧都在变，
// 遮罩只跟 pull 有关（0.28 → 0），没必要每次都让根视图重新构造这一段。
// 压暗峰值 4.9.6 由 0.42 收到 0.28：配合缩放从 6% 收到 3%，下拉时只是「略微退后」，
// 不再像关灯一样整屏变黑。
//
// 注意它仍然挂在 TabView 的 `.overlay` 上；打开 / 关闭播放页时这层压暗的渐变
// 用与外层同一条 spring（0.46 / 0.90），所以视觉和抽取之前完全一样。
private struct DimOverlay: View {
    @ObservedObject var player = PlayerPresenter.shared

    var body: some View {
        Color.black
            .opacity(player.isOpen ? 0.28 * (1 - Double(player.pull)) : 0)
            // 和外层那圈完全同一条 spring（0.46 / 0.90），只是在这里再写一遍：
            // 这层现在是通过自己的订阅收到 isOpen 变化的，不保证一定落在
            // RootTabView 那圈 `.animation(…, value: player.isOpen)` 的事务里，
            // 少了它就可能变成硬切。参数一模一样，视觉上没有任何差别。
            // 注意 value 只看 isOpen —— 拖动时 opacity 跟 pull 逐帧变化，不能被动画带。
            .animation(.spring(response: 0.46, dampingFraction: 0.90), value: player.isOpen)
            .ignoresSafeArea()
            .allowsHitTesting(false)
    }
}
