import SwiftUI
import UIKit

// MARK: - 安全区
//
// 播放页要铺满整屏（含状态栏 / 底部横条区域），所以它自己忽略安全区，
// 再按真实安全区把内容往里让 —— 否则 `.clipShape` 会按安全区内的 frame 裁剪，
// 上下各留一条黑边（4.7 的「重大 Bug」就是这么来的）。

enum SafeArea {
    /// 取不到窗口时的回退值：按灵动岛机型（top 59 / bottom 34）。
    /// 绝不能用 0 —— 0 会让播放页内容顶到状态栏和底部横条下面。
    private static let fallback = UIEdgeInsets(top: 59, left: 0, bottom: 34, right: 0)

    static var insets: UIEdgeInsets {
        let scenes = UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }
        // 优先前台活跃的场景，没有再退回任意一个
        let scene = scenes.first(where: { $0.activationState == .foregroundActive }) ?? scenes.first
        let win = scene?.windows.first(where: { $0.isKeyWindow }) ?? scene?.windows.first
        // 首帧可能还没布局：top == 0 视为「拿不到」，回退到灵动岛基准值
        guard let insets = win?.safeAreaInsets, insets.top > 0 else { return fallback }
        return insets
    }
}

// MARK: - 播放页的呈现方式
//
// 以前用 fullScreenCover：那一层是完全独立的窗口层，往下拖时下面什么都没有，
// 露出来就是大片黑色。改成在根视图里叠一张自己管理的卡片：
// 打开时底下的 TabView 稍微缩小并压暗，播放页从底部推上来，
// 往下拖的时候底下的菜单一直看得见 —— 就是 iOS 那种「回到上一级」的观感。

final class PlayerPresenter: ObservableObject {
    static let shared = PlayerPresenter()
    @Published var isOpen = false
    /// 下拉进度：0 = 完全打开，1 = 已经拉到底（松手就关）。
    /// 下层菜单靠它跟着渐进放大 / 变亮 / 变圆角，而不是等关闭动画才动一下。
    @Published var pull: CGFloat = 0
    /// 播放页开着时把整个 App 切到深色（状态栏文字才会转白）。
    /// 不能随手写 `.preferredColorScheme(.dark)`：那会在播放页还在滑入、菜单露在外面的
    /// 那 0.4 秒里把整个窗口换成深色，看起来就是一整块颜色闪一下。
    /// 所以改成「完全盖住之后再切、开始下滑之前就切回来」——两次切换都发生在被遮住的时候。
    @Published var forceDark = false

    private init() {}

    func open() {
        guard !isOpen else { return }
        pull = 0
        withAnimation(.spring(response: 0.46, dampingFraction: 0.90)) {
            isOpen = true
        }
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.52) { [weak self] in
            // 0.52s 内用户已经把它关掉了：不要再把窗口切成深色，
            // 否则下一次打开会在「已经盖满」之前就变成深色，露出来那一下就是闪。
            guard let self = self, self.isOpen else { return }
            self.forceDark = true
        }
    }

    /// 往下拖的**第一帧**：这时位移只有 8pt 左右，底下几乎还没露出来，
    /// 在这里把窗口切回浅色谁也看不见；再晚一点（比如松手时才切），
    /// 底下已经露出的那整页会当场由深变浅 —— 那就是「下拉时闪一下」的来源。
    /// 用关掉动画的 Transaction：preferredColorScheme 的变化一旦被动画带，
    /// 会变成一段 0.4s 的渐变，和手势叠在一起看更花。
    func beginDismiss() {
        guard forceDark else { return }
        var t = Transaction()
        t.disablesAnimations = true
        withTransaction(t) { forceDark = false }
    }

    /// 没拖够、弹回原位：等弹回动画（response 0.34）走完再把窗口切回深色。
    /// 回调里必须复查状态 —— 这 0.36s 内用户可能又把它关掉了，或者又开始拖了。
    func cancelDismiss() {
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.36) { [weak self] in
            guard let self = self, self.isOpen, self.pull == 0 else { return }
            self.forceDark = true
        }
    }

    func close() {
        guard isOpen else { return }
        pull = 0
        // 先把窗口切回浅色。此刻播放页还盖满全屏，切了看不见；
        // 用 disablesAnimations 的 Transaction，理由同 beginDismiss。
        let needFlip = forceDark
        if needFlip {
            var t = Transaction()
            t.disablesAnimations = true
            withTransaction(t) { forceDark = false }
            // 让配色切换先单独渲染一帧（这一帧播放页仍盖满屏），0.05s 后再下滑
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.05) { [weak self] in
                self?.startDismissAnimation()
            }
        } else {
            // 拖动开始时已经切过浅色了，不用再等
            startDismissAnimation()
        }
    }

    private func startDismissAnimation() {
        withAnimation(.spring(response: 0.40, dampingFraction: 0.92)) {
            isOpen = false
        }
    }
}

// MARK: - 训练面板
//
// 从播放页的训练卡点进来。内容就是训练页那套（同一个 Store / RestTimer），
// 只是排得更紧凑，边听歌边记组不用跳回训练页。

struct TrainingSheet: View {
    @ObservedObject var store = Store.shared
    @ObservedObject var rest = RestTimer.shared
    @Environment(\.dismiss) private var dismiss
    @State private var detent: PresentationDetent = .large

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 16) {
                    exerciseCard

                    if rest.isRunning { restCard }

                    HStack(spacing: 12) {
                        StepperField(label: "重量", unit: "kg",
                                     value: formatWeight(store.lastWeight),
                                     onMinus: { store.setWeight(store.lastWeight - 2.5) },
                                     onPlus: { store.setWeight(store.lastWeight + 2.5) })
                        StepperField(label: "次数", unit: "reps",
                                     value: "\(store.lastReps)",
                                     onMinus: { store.setReps(store.lastReps - 1) },
                                     onPlus: { store.setReps(store.lastReps + 1) })
                    }

                    Button {
                        completeSet()
                    } label: {
                        Text("完成一组")
                    }
                    .buttonStyle(GlassButtonStyle(prominent: true))

                    setListCard

                    if store.current != nil {
                        Button(role: .destructive) {
                            rest.skip()
                            store.finishSession()
                            dismiss()
                        } label: {
                            Text("结束本次训练")
                                .font(.system(size: 14, weight: .medium))
                                .frame(maxWidth: .infinity)
                                .frame(height: 44)
                        }
                        .buttonStyle(.plain)
                        .foregroundColor(.red)
                    }
                }
                .padding(.horizontal, 20)
                .padding(.top, 8)
                .padding(.bottom, 26)
            }
            .background { AppBackground() }
            .scrollDismissesKeyboard(.immediately)
            .navigationTitle("训练")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button("完成") { dismiss() }
                        .fontWeight(.semibold)
                }
            }
            .onAppear { store.ensureSession() }
        }
        .presentationDetents([.medium, .large], selection: $detent)
        .presentationDragIndicator(.visible)
    }

    // MARK: 当前动作

        return VStack(alignment: .leading, spacing: 8) {

                .font(.system(size: 11, weight: .semibold))
                .foregroundColor(.secondary)

                .font(.system(size: 20, weight: .bold))
                .lineSpacing(2)
                .lineLimit(1)
            HStack(spacing: 6) {
                Text("已做 \(ex.sets.count) 组")
                    .font(.system(size: 12))
                    .lineSpacing(2)
                    .foregroundColor(.secondary)
                if let s = ex.sets.last {
                    Text("· 上一组 \(formatWeight(s.w)) kg × \(s.r)")
                        .font(.system(size: 12))
                        .lineSpacing(2)
                        .foregroundColor(.secondary)
                }
            }
        }
        .frame(maxWidth: .infinity, alignment: .leading)
        .padding(18)
        .glass(cornerRadius: 20, shadow: false)
    }

    // MARK: 休息

    private var restCard: some View {
        HStack(spacing: 14) {
            MiniRestRing(progress: rest.progress, text: rest.remainingText, size: 58)
            VStack(alignment: .leading, spacing: 5) {
                Text("组间休息")
                    .font(.system(size: 13, weight: .semibold))
                Text("结束后会自动提示")
                    .font(.system(size: 11))
                    .foregroundColor(.secondary)
            }
            Spacer(minLength: 4)
            Button {
                rest.add(seconds: 30)
            } label: {
                Text("+30")
                    .font(.system(size: 12, weight: .semibold))
                    .frame(width: 48, height: 32)
                    .background(Capsule().fill(Color.primary.opacity(0.08)))
            }
            .buttonStyle(ScaleButtonStyle())
            Button {
                rest.skip()
            } label: {
                Text("跳过")
                    .font(.system(size: 12, weight: .medium))
                    .foregroundColor(.secondary)
                    .frame(width: 44, height: 32)
            }
            .buttonStyle(ScaleButtonStyle())
        }
        .padding(16)
        .glass(cornerRadius: 20, shadow: false)
    }

    // MARK: 组列表

    private var setListCard: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("本次记录")
                .font(.system(size: 12, weight: .semibold))
                .foregroundColor(.secondary)

            let sets = store.currentExercise.sets
            if sets.isEmpty {
                Text("还没有记录，调好重量和次数点「完成一组」")
                    .font(.system(size: 12))
                    .lineSpacing(3)
                    .foregroundColor(.secondary)
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding(.vertical, 6)
            } else {
                ForEach(Array(sets.enumerated()), id: \.element.id) { i, s in
                    HStack(spacing: 10) {
                        Text("\(i + 1)")
                            .font(.system(size: 11, weight: .semibold, design: .rounded))
                            .foregroundColor(.secondary)
                            .frame(width: 18, alignment: .trailing)
                        Text("\(formatWeight(s.w)) kg")
                            .font(.system(size: 14, weight: .medium, design: .rounded))
                        Text("× \(s.r)")
                            .font(.system(size: 13, design: .rounded))
                            .foregroundColor(.secondary)
                        Spacer(minLength: 4)
                        Button {
                            store.deleteSet(s.id)
                        } label: {
                            Image(systemName: "minus.circle")
                                .font(.system(size: 15))
                                .foregroundColor(.secondary)
                        }
                        .buttonStyle(.plain)
                    }
                    .padding(.vertical, 6)
                }
            }
        }
        .frame(maxWidth: .infinity, alignment: .leading)
        .padding(18)
        .glass(cornerRadius: 20, shadow: false)
    }

    private func completeSet() {
        _ = store.ensureSession()
        store.logSet(weight: store.lastWeight, reps: store.lastReps)
        UIImpactFeedbackGenerator(style: .medium).impactOccurred()
        rest.start(seconds: store.restSeconds)
    }

    private func formatWeight(_ w: Double) -> String {
        w.truncatingRemainder(dividingBy: 1) == 0
            ? String(format: "%.0f", w)
            : String(format: "%.1f", w)
    }
}

import SwiftUI

// MARK: - 音乐
//
// 参照 Apple Music 的极简结构：大标题 + 常驻搜索栏 + 列表 + 底部迷你播放条，
// 点开是铺满全屏的播放页。搜索直接查 Apple Music 曲库，点歌在 App 内播放，
// 不再把用户甩到别的 App。

struct MusicView: View {
    @ObservedObject var hub = MusicHub.shared
    @ObservedObject var store = Store.shared
    @ObservedObject var am = AppleMusicService.shared
    @ObservedObject var player = PlayerPresenter.shared

    @State private var query = ""
    /// 资料库搜索结果（全部可直接完整播放，不再有 30 秒试听）
    @State private var libraryHits: [MPMediaItem] = []
    /// 命中的专辑（按专辑排版用，带整张曲目）
    @State private var albumHits: [AppleMusicService.Album] = []
    @State private var didSearch = false
    @State private var showPicker = false

    /// 音乐页的两个界面：Apple Music 曲库 / 本地音乐
    enum SourceTab: String, CaseIterable, Identifiable {
        case apple = "Apple Music"
        case local = "本地"
        var id: String { rawValue }
    }

    @State private var tab: SourceTab = .apple
    /// 卡片进度条拖动中的临时秒数
    @State private var cardScrub: Double?

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(alignment: .leading, spacing: 14) {
                    Picker("来源", selection: $tab) {
                        ForEach(SourceTab.allCases) { t in
                            Text(t.rawValue).tag(t)
                        }
                    }
                    .pickerStyle(.segmented)

                    switch tab {
                    case .apple:
                        applePane
                    case .local:
                        localPane
                    }
                }
                .padding(.horizontal, 20)
                .padding(.top, 6)
                .padding(.bottom, 130)
            }
            .background { AppBackground() }
            .scrollDismissesKeyboard(.interactively)
            .searchable(text: $query,
                        placement: .navigationBarDrawer(displayMode: .always),
                        prompt: Text(tab == .apple ? "搜索我的资料库" : "搜索本地音乐"))
            .onChange(of: query) { _ in
                runLibrarySearch()
            }
            .onChange(of: am.librarySongCount) { _ in
                runLibrarySearch()
            }
            .onChange(of: tab) { _ in
                if tab == .apple { runLibrarySearch() }
            }
            .navigationTitle("音乐")
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    Menu {
                        Button {
                            showPicker = true
                        } label: {
                            Label("导入本地音乐", systemImage: "square.and.arrow.down")
                        }
                        Button {
                            am.buildIndexIfNeeded(force: true)
                        } label: {
                            Label("刷新 Apple Music 资料库", systemImage: "arrow.clockwise")
                        }
                        Button {
                            NeteaseService.openApp()
                        } label: {
                            Label("打开网易云音乐", systemImage: "waveform")
                        }
                    } label: {
                        Image(systemName: "ellipsis.circle")
                            .font(.system(size: 17))
                    }
                }
            }
            .safeAreaInset(edge: .bottom) {
                if hub.current != nil, !player.isOpen {
                    MiniPlayerBar { player.open() }
                }
            }
            .navigationDestination(for: String.self) { key in
                AlbumDetailView(albumKey: key)
            }
            .sheet(isPresented: $showPicker) {
                DocumentPicker(contentTypes: [.audio], allowsMultiple: true) { urls in
                    let n = hub.importFiles(urls)
                    if n > 0, hub.localTracks.count == n {
                        hub.playLocal(at: 0)
                    }
                }
            }
        }
    }

    // MARK: - 两个界面

    @ViewBuilder
    private var applePane: some View {
        // 搜索页不放播放器卡片、不放「已连接」卡片 —— 底下已经有迷你播放条了。
        // 只有还没授权时才留一行极简提示，否则搜不出任何东西。
        if !am.isGranted { authRow }

        if query.trimmingCharacters(in: .whitespaces).isEmpty {
            searchHint
        } else {
            libraryResults
        }
    }

    @ViewBuilder
    private var localPane: some View {
        if hub.current != nil, hub.current?.source == .local {
            nowPlayingCard
        }
        localSection
        if let err = hub.errorMessage {
            Text(err)
                .font(.system(size: 11))
                .foregroundColor(.red)
        }
    }

    private var searchHint: some View {
        VStack(spacing: 8) {
            Image(systemName: "magnifyingglass")
                .font(.system(size: 24, weight: .light))
                .foregroundColor(.secondary)
            Text("搜我的资料库")
                .font(.system(size: 15, weight: .semibold))
            Text("搜歌名、艺人、专辑，也可以打拼音首字母。想加歌就去 Apple Music 里点「+」。")
                .font(.system(size: 11))
                .foregroundColor(.secondary)
                .multilineTextAlignment(.center)
                .fixedSize(horizontal: false, vertical: true)
        }
        .frame(maxWidth: .infinity)
        .padding(.vertical, 26)
        .padding(.horizontal, 18)
    }

    // MARK: - 未授权时的一行提示

    @ViewBuilder
    private var authRow: some View {
        HStack(spacing: 8) {
            Image(systemName: "music.note.tv")
                .font(.system(size: 13, weight: .medium))
                .foregroundColor(.accentColor)
            Text(am.access == .denied ? "资料库权限被关闭，搜不到歌" : "需要读取你的 Apple Music 资料库")
                .font(.system(size: 12))
                .foregroundColor(.secondary)
                .lineLimit(1)
            Spacer(minLength: 4)
            Button {
                if am.access == .denied {
                    if let u = URL(string: UIApplication.openSettingsURLString) {
                        UIApplication.shared.open(u)
                    }
                } else {
                    am.requestAccess { ok in
                        if ok { runLibrarySearch() }
                    }
                }
            } label: {
                Text(am.access == .denied ? "去设置" : "授权")
                    .font(.system(size: 12, weight: .semibold))
                    .foregroundColor(.accentColor)
                    .padding(.horizontal, 12)
                    .frame(height: 30)
                    .background(Capsule().fill(Color.accentColor.opacity(0.13)))
            }
            .buttonStyle(ScaleButtonStyle())
        }
        .padding(.horizontal, 4)
    }

    // MARK: - 资料库搜索
    //
    // 只搜自己的 Apple Music 资料库：不联网、不搜曲库以外的任何东西。
    // 排序和相关性门限都在 AppleMusicService.search 里做（多词 AND、按分数、
    // 支持拼音），这里只负责调用 + 顺手预热前几行的封面。

    private func runLibrarySearch() {
        let q = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard tab == .apple, am.isGranted, !q.isEmpty else {
            libraryHits = []
            albumHits = []
            didSearch = false
            return
        }
        didSearch = true
        libraryHits = am.search(q)
        albumHits = am.albums(from: libraryHits, matching: q)
        // 前几行的封面先排进后台，滚动时不至于一片灰
        ArtworkStore.shared.prefetch(libraryHits, side: 160, limit: 12)
    }

    // MARK: - 搜索结果（专辑网格 + 歌曲）
    //
    // 照 Apple Music 的版式：先「专辑」两列网格（封面 / 专辑名 / 艺人·年份），
    // 点进一张专辑能看到**整张**曲目（不再只有命中的那几首），下面是「歌曲」直接开播。
    // 队列仍是整串结果，persistentID 精确寻址，点哪首就是哪首。

    private var tileSide: CGFloat {
        let w = UIScreen.main.bounds.width - 40      // 页面左右各 20
        return max(120, floor((w - 12) / 2))         // 两列，列间距 12
    }

    @ViewBuilder
    private var libraryResults: some View {
        if am.isIndexing {
            HStack(spacing: 8) {
                ProgressView().scaleEffect(0.7)
                Text("正在读取资料库…")
                    .font(.system(size: 12))
                    .foregroundColor(.secondary)
                Spacer()
            }
            .padding(.vertical, 16)
        } else if libraryHits.isEmpty {
            if didSearch {
                Text("资料库里没有「\(query)」")
                    .font(.system(size: 12))
                    .foregroundColor(.secondary)
                    .frame(maxWidth: .infinity)
                    .padding(.vertical, 20)
            }
        } else {
            if !albumHits.isEmpty {
                sectionHeader("专辑", trailing: "\(albumHits.count) 张")

                LazyVGrid(columns: [GridItem(.flexible(), spacing: 12),
                                    GridItem(.flexible(), spacing: 12)],
                          alignment: .leading,
                          spacing: 18) {
                    ForEach(albumHits) { a in
                        albumTile(a)
                    }
                }
                .padding(.bottom, 6)
            }

            sectionHeader("歌曲", trailing: "\(libraryHits.count) 首")

            LazyVStack(spacing: 0) {
                ForEach(libraryHits.indices, id: \.self) { i in
                    libraryRow(libraryHits[i], index: i)
                    if i != libraryHits.count - 1 {
                        Divider().padding(.leading, 66)
                    }
                }
            }
            .padding(.vertical, 4)
            .glass()
        }
    }

    private func sectionHeader(_ title: String, trailing: String) -> some View {
        HStack(spacing: 6) {
            Text(title)
                .font(.system(size: 17, weight: .bold))
            Spacer(minLength: 4)
            Text(trailing)
                .font(.system(size: 12))
                .foregroundColor(.secondary)
        }
        .padding(.horizontal, 4)
        .padding(.top, 4)
    }

    /// 专辑格子：封面 + 标题 + 艺人·年份（点进去是整张专辑）
    private func albumTile(_ a: AppleMusicService.Album) -> some View {
        NavigationLink(value: a.id) {
            VStack(alignment: .leading, spacing: 8) {
                TrackArt(title: a.title, mediaID: a.sampleID, size: tileSide, corner: 6)
                    .shadow(color: .black.opacity(0.10), radius: 8, y: 4)

                Text(a.title)
                    .font(.system(size: 14, weight: .semibold))
                    .foregroundColor(.primary)
                    .lineLimit(1)

                Text(a.subtitle)
                    .font(.system(size: 12))
                    .foregroundColor(.secondary)
                    .lineLimit(1)
                    .padding(.top, 2)
            }
            .frame(width: tileSide, alignment: .leading)
            .contentShape(Rectangle())
        }
        .buttonStyle(.plain)
    }

    /// 歌曲行：封面 + 歌名 + 艺人·专辑 + 时长
    private func libraryRow(_ m: MPMediaItem, index: Int) -> some View {
        let playing = hub.current?.mediaID == m.persistentID
        return Button {
            playLibrary(at: index)
        } label: {
            HStack(spacing: 12) {
                TrackArt(title: m.title ?? "未知曲目",
                         mediaID: m.persistentID,
                         size: 50)
                VStack(alignment: .leading, spacing: 4) {
                    Text(m.title ?? "未知曲目")

                        .foregroundColor(.secondary)

                .padding(.vertical, 60)

                MiniPlayerBar { player.open() }

        }

    // MARK: 头部

    private func header(_ a: AppleMusicService.Album) -> some View {
        VStack(spacing: 16) {
            TrackArt(title: a.title, mediaID: a.sampleID, size: coverSide, corner: 8)
                .shadow(color: .black.opacity(0.16), radius: 18, y: 10)

            VStack(spacing: 6) {
                // Apple Music 的顺序：大字是**专辑名**，小字才是「艺人 · 年份」

                    .font(.system(size: 21, weight: .bold))
                    .lineSpacing(3)
                    .multilineTextAlignment(.center)
                    .lineLimit(2)

                    .lineSpacing(2)

            .padding(.horizontal, 24)
        }
        .frame(maxWidth: .infinity)
        .padding(.top, 16)
    }

    private func actions(_ a: AppleMusicService.Album) -> some View {
        HStack(spacing: 12) {
            Button {
                play(a, shuffled: false)
            } label: {
                Label("播放", systemImage: "play.fill")
            }
            .buttonStyle(GlassButtonStyle())

                play(a, shuffled: true)
            } label: {

            }
            .buttonStyle(GlassButtonStyle())
        }
        .padding(.horizontal, 20)
        .padding(.top, 22)
        .padding(.bottom, 18)
    }

    // MARK: 曲目

    private func trackList(_ a: AppleMusicService.Album) -> some View {
        VStack(spacing: 0) {
            ForEach(a.items.indices, id: \.self) { i in
                trackRow(a.items[i], position: i + 1, album: a, index: i)
                if i != a.items.count - 1 {
                    Divider().padding(.leading, 56)
                }
            }
        }
        .padding(.vertical, 6)
        .glass()
        .padding(.horizontal, 20)
    }

    private func trackRow(_ m: MPMediaItem,
                          position: Int,
                          album a: AppleMusicService.Album,
                          index: Int) -> some View {
        let playing = hub.current?.mediaID == m.persistentID
        let no = m.albumTrackNumber > 0 ? m.albumTrackNumber : position
        return Button {
            play(a, shuffled: false, startAt: index)
        } label: {
            HStack(spacing: 10) {
                Text("\(no)")
                    .font(.system(size: 12, design: .rounded))
                    .monospacedDigit()
                    .foregroundColor(playing ? .accentColor : .secondary)
                    .frame(width: 22, alignment: .trailing)

                Text(m.title ?? "未知曲目")
                    .font(.system(size: 15, weight: playing ? .semibold : .regular))
                    .lineSpacing(3)
                    .foregroundColor(playing ? .accentColor : .primary)
                    .lineLimit(2)

                Spacer(minLength: 6)

                if playing && hub.isPlaying {
                    Image(systemName: "waveform")
                        .font(.system(size: 13))
                        .foregroundColor(.accentColor)
                } else {
                    Text(MusicHub.formatTime(m.playbackDuration))
                        .font(.system(size: 11))
                        .monospacedDigit()
                        .foregroundColor(.secondary)
                }
            }
            .padding(.leading, 12)
            .padding(.trailing, 14)
            .padding(.vertical, 13)
            .frame(minHeight: 52)
            .contentShape(Rectangle())
        }
        .buttonStyle(.plain)
    }

    // MARK: 播放

    private func play(_ a: AppleMusicService.Album, shuffled: Bool, startAt: Int = 0) {
        var media = a.items
        var start = startAt
        if shuffled {
            media.shuffle()
            start = 0
        }
        let items = MusicHub.items(from: media)
        guard items.indices.contains(start) else { return }
        hub.play(items: items, startingAt: start)
        player.open()
    }
}

// MARK: - 全屏播放页
//
// 版式跟 Apple Music 对齐（左边距统一 24，所有元素共用一条中轴）：
//
//            ── 抓手 ──
//   ⌄          正在播放           ⋯
//        [      封面      ]      屏宽 76%，上限 322
//   歌名（22 粗，左对齐）
//   艺人（17，灰，左对齐）
//   ══════ 进度条 8pt ══════
//   0:12                  -3:20
//   ☰     ⏮     ▶     ⏭     🔁
//   🔈 ═════ 音量 ═════ 🔊
//   [ 训练状态卡 ]  ← 点开是训练面板
//
// 背景是灰色液态玻璃，整页固定深色，所以进度 / 音量条是白色。

struct NowPlayingView: View {
    @ObservedObject var hub = MusicHub.shared
    @ObservedObject var store = Store.shared
    @ObservedObject var rest = RestTimer.shared
    @ObservedObject var player = PlayerPresenter.shared

    @State private var dragOffset: CGFloat = 0
    /// 这一轮下拉有没有已经通知过「开始关闭」（只在第一次 onChanged 里发一次）
    @State private var dismissing = false
    @State private var showQueue = false
    @State private var showTraining = false
    /// 拖动进度条时的临时秒数（nil = 没在拖，跟随播放器）
    @State private var scrub: Double?

    /// 全页统一左右留白：封面、标题、进度条、按钮、音量都对齐到这条线
    private let inset: CGFloat = 24

    var body: some View {
        GeometryReader { geo in
            let safe = SafeArea.insets
            // 除封面以外的固定高度（抓手 + 头部 + 标题 + 进度 + 控制 + 音量 + 训练卡）
            //
            // 4.9.2 由 448 加到 462，构成：标题区两行间距 +3（2→5）、
            // 训练卡文字间距 +4（4→6，两处间隙各 +2）、训练卡上下内边距 +4（12→14）、
            // 右侧三个小按钮间距 +2（8→9）、再留 1pt 余量。
            // 小屏预算（side = min(宽×0.82, 高 - 上下安全区 - 462, 340)）：
            //   iPhone SE3 375×667（安全区 20/0）→ 185pt，未触及 130 下限；
            //   iPhone 13 mini 375×812（50/34）→ 266pt；iPhone 16（59/34）→ 297pt。
            // 也就是说小屏上封面会变小一点，但不会被挤出屏幕（Spacer 还能再压）。
            let fixed: CGFloat = 462
            // 铺满整屏之后高度富余，封面按屏宽 82% 走，上限 340
            let side = max(130, min(geo.size.width * 0.82,
                                    geo.size.height - safe.top - safe.bottom - fixed,
                                    340))
            // 标题 / 进度 / 控制 / 音量都对齐到封面的左右边线（Apple Music 就是这个对齐）
            let pad = max(inset, (geo.size.width - side) / 2)

            VStack(spacing: 0) {
                grabber

                header

                Spacer(minLength: 10)

                cover(side: side)

                Spacer(minLength: 18)
                    .frame(height: 24)

                titleBlock(pad: pad)

                Spacer(minLength: 16)
                    .frame(height: 20)

                progressRow(pad: pad)

                Spacer(minLength: 12)
                    .frame(height: 18)

                controls(pad: pad)

                Spacer(minLength: 12)
                    .frame(height: 18)

                volume(pad: pad)

                Spacer(minLength: 10)

                trainingBar
                    .padding(.horizontal, 16)
            }
            // 忽略安全区再自己让回来：这样背景和圆角裁剪都覆盖到状态栏 / 底部横条，
            // 不会在上下各留一条黑边
            .padding(.top, safe.top)
            .padding(.bottom, max(safe.bottom, 10))
            .frame(width: geo.size.width, height: geo.size.height)
        }
        .ignoresSafeArea()
        .background { playerBackground }
        // 顺序必须是「背景 → CardClip → offset」：
        // 遮罩要盖在卡片本身上，再让整张卡片往下位移，圆角才会跟着走；
        // 反过来（先 offset 再 mask）遮罩会留在原地，拖动时圆角跑到屏幕角落去。
        .modifier(CardClip(active: dragOffset > 0.5, radius: min(34, dragOffset / 5)))
        .offset(y: max(0, dragOffset))
        .environment(\.colorScheme, .dark)
        .sheet(isPresented: $showQueue) {
            QueueSheet()
        }
        .sheet(isPresented: $showTraining) {
            TrainingSheet()
        }
    }

    /// 灰色液态玻璃：灰底 + 半透明材质 + 中性柔光 + 薄白釉面。
    /// 只有灰度没有色相，所以不会发蓝发紫。

                .fill(Color.white.opacity(0.05))

            } label: {

                    .font(.system(size: 17, weight: .semibold))

        .contentShape(Rectangle())
        .gesture(dismissDrag)
    }

    /// 往下拖收回：底下的菜单一直可见（并被一起带着放大 / 变亮），所以不会出现黑边；
    /// 拖过阈值就关，没拖够就弹回原位。
    private var dismissDrag: some Gesture {
        DragGesture(minimumDistance: 8)
            .onChanged { v in
                let h = max(0, v.translation.height)
                dragOffset = h
                // 0…1：拉到 260pt 就算到底，下层同步动
                player.pull = min(1, h / 260)
                // 刚拖起来（位移还只有 8pt 左右、底下几乎没露出）就把窗口切回浅色。
                // 这一步不能等到松手：拖到一半时底下那整页已经露出来了，
                // 那时再从深变浅，就是肉眼看到的「下拉闪一下」。
                if h > 0 && !dismissing {
                    dismissing = true
                    player.beginDismiss()
                }
            }
            .onEnded { v in
                dismissing = false
                let far = v.translation.height > 110 || v.predictedEndTranslation.height > 260
                if far {
                    player.close()
                } else {
                    withAnimation(.spring(response: 0.34, dampingFraction: 0.86)) {
                        dragOffset = 0
                        player.pull = 0
                    }
                    // 弹回原位后再把窗口切回深色（0.36s 后，弹回动画 0.34s）
                    player.cancelDismiss()
                }
            }
    }

// MARK: - 玻璃进度条 / 音量条
//
// 照 Apple Music 的观感做：8pt 厚的胶囊条，浅灰半透明轨道 +
// 已播放段是白色（深色模式）/ 近黑（浅色模式），0.5pt 高光描边 + 轻微外发光，
// 看起来像一根磨过的玻璃条，而不是系统 Slider 那种塑料件。
// 没有圆点把手（按住任意位置拖，命中区 26pt），拖动时按下位置跟手。

struct GlassBar: View {
    var progress: Double
    var onScrub: (Double) -> Void = { _ in }
    var onCommit: (Double) -> Void = { _ in }
    var trackHeight: CGFloat = 8
    var enabled: Bool = true

    @Environment(\.colorScheme) private var scheme
    @State private var dragging = false
    @State private var local: Double = 0

    var body: some View {
        GeometryReader { geo in
            let w = max(1, geo.size.width)
            let p = CGFloat(min(max(dragging ? local : progress, 0), 1))

            ZStack(alignment: .leading) {
                // 轨道：玻璃槽
                Capsule(style: .continuous)
                    .fill(Color.primary.opacity(scheme == .dark ? 0.22 : 0.12))
                    .overlay(
                        Capsule(style: .continuous)
                            .strokeBorder(Color.white.opacity(scheme == .dark ? 0.12 : 0.45),
                                          lineWidth: 0.5)
                    )
                    .frame(height: trackHeight)

                // 已播放段：白 / 深灰玻璃
                Capsule(style: .continuous)
                    .fill(fillColor)
                    .overlay(
                        Capsule(style: .continuous)
                            .strokeBorder(Color.white.opacity(scheme == .dark ? 0.65 : 0.30),
                                          lineWidth: 0.5)
                    )
                    .frame(height: trackHeight)
                    .mask(alignment: .leading) {
                        Rectangle().frame(width: w * p)
                    }
                    .shadow(color: fillColor.opacity(scheme == .dark ? 0.45 : 0.35), radius: 4, y: 0)
            }
            .frame(width: w, height: max(trackHeight, 26))
            .contentShape(Rectangle())
            .gesture(
                DragGesture(minimumDistance: 0)
                    .onChanged { v in
                        guard enabled else { return }
                        if !dragging {
                            dragging = true
                            UISelectionFeedbackGenerator().selectionChanged()
                        }
                        let t = min(max(v.location.x / w, 0), 1)
                        local = t
                        onScrub(t)
                    }
                    .onEnded { v in
                        guard enabled else { return }
                        let t = min(max(v.location.x / w, 0), 1)
                        local = t
                        dragging = false
                        onCommit(t)
                    }
            )
        }
        .frame(height: max(trackHeight, 26))
        .opacity(enabled ? 1 : 0.4)
    }

    private var fillColor: Color {
        scheme == .dark
            ? Color.white.opacity(0.97)
            : Color(white: 0.12).opacity(0.86)
    }
}

// MARK: - 系统音量条

struct VolumeBar: View {
    var trackHeight: CGFloat = 8

    @State private var level: Double = SystemVolume.shared.level
    @State private var dragging = false
    @State private var local: Double = 0

    private let ticker = Timer.publish(every: 0.25, on: .main, in: .common).autoconnect()

    var body: some View {
        GlassBar(progress: dragging ? local : level,
                 onScrub: { v in
                     dragging = true
                     local = v
                     SystemVolume.shared.set(v)
                 },
                 onCommit: { v in
                     local = v
                     level = v
                     dragging = false
                     SystemVolume.shared.set(v)
                 },
                 trackHeight: trackHeight)
            .onReceive(ticker) { _ in
                guard !dragging else { return }
                let v = SystemVolume.shared.level
                if abs(v - level) > 0.005 { level = v }
            }
            .onAppear { level = SystemVolume.shared.level }
    }
}

/// 系统音量读写。
///
/// iOS 不开放改音量的 API，只能借 MPVolumeView 内部那个 UISlider：
/// 把它挂到一个看不见的容器里，读它的 value / 给它 setValue，
/// 就能做到和侧边键完全同步的真实音量控制。
final class SystemVolume {
    static let shared = SystemVolume()

    private let host = MPVolumeView(frame: CGRect(x: -4, y: -4, width: 1, height: 1))

    private var slider: UISlider? {
        host.subviews.compactMap { $0 as? UISlider }.first
    }

    private init() {
        host.showsRouteButton = false
        host.alpha = 0.01
        host.isUserInteractionEnabled = false
    }

    /// MPVolumeView 的 UISlider 是懒创建的，先进窗口再取一次。
    private func install() {
        if host.superview == nil {
            guard let scene = UIApplication.shared.connectedScenes
                    .compactMap({ $0 as? UIWindowScene })
                    .first(where: { $0.activationState == .foregroundActive })
                    ?? UIApplication.shared.connectedScenes.compactMap({ $0 as? UIWindowScene }).first,
                  let window = scene.windows.first(where: { $0.isKeyWindow }) ?? scene.windows.first
            else { return }
            window.addSubview(host)
            host.setNeedsLayout()
            host.layoutIfNeeded()
        }
        if slider == nil {
            host.setNeedsLayout()
            host.layoutIfNeeded()
        }
    }

    var level: Double {
        install()
        if let s = slider { return Double(s.value) }
        return Double(AVAudioSession.sharedInstance().outputVolume)
    }

    func set(_ v: Double) {
        install()
        let target = Float(min(max(v, 0), 1))
        if let s = slider {
            s.setValue(target, animated: false)
        }
    }
}

/// 本地文件没有内嵌封面时，用歌名挑一个纯色占位（不用渐变）。
struct AlbumArt: View {
    let title: String
    var size: CGFloat

    var body: some View {
        ZStack {
            RoundedRectangle(cornerRadius: size * 0.08, style: .continuous)
                .fill(Self.baseColor(for: title))
            Image(systemName: "music.note")
                .font(.system(size: size * 0.2, weight: .light))
                .foregroundColor(.white.opacity(0.92))
        }
        .frame(width: size, height: size)
    }

    static func baseColor(for title: String) -> Color {
        let colors: [Color] = [
            Color(red: 0.32, green: 0.42, blue: 0.92),
            Color(red: 0.92, green: 0.44, blue: 0.38),
            Color(red: 0.16, green: 0.66, blue: 0.56),
            Color(red: 0.80, green: 0.36, blue: 0.58),
            Color(red: 0.24, green: 0.60, blue: 0.82)
        ]
        let seed = title.unicodeScalars.reduce(0) { $0 &+ Int($1.value) }
        return colors[abs(seed) % colors.count]
    }
}

/// 底部栏用系统原生的 TabView（UIKit 的 UITabBar）：
/// 点按回弹、长按、滚动到底部再点当前页回顶、无障碍朗读、
/// 安全区处理和玻璃材质全部交给系统，新版 iOS 上自动是新外观。
struct RootTabView: View {
    @Environment(\.scenePhase) private var scenePhase
    @ObservedObject var player = PlayerPresenter.shared
    @State private var selection = 0

    var body: some View {
        ZStack {
            // 底层：常态用「页面同色」，不是黑色。
            // TabView 和它里面的页面都是按安全区布局的（系统在根视图里已经替它们让开了
            // 状态栏与底部横条）。只要这一层是黑的，任何一处没被内容盖住的安全区就会露出黑边，
            // 所以这里铺的必须是跟页面一样的底色 —— 露出来也看不出来。
            AppBackground()

            // 播放页打开时这一层换成播放页自己的底色（深灰 #212121），当卡片堆叠的底。
            // 用播放页同色而不是纯黑：万一播放页边缘有一点没铺到，露出来的也是同一个颜色。
            // 淡入用 0.25s easeInOut：和 TabView 那 0.46s 的 spring 大致同步，
            // 避免 forceDark 已经切了深色、这一层还是浅灰的那一帧硬闪。
            Color(white: 0.13)
                .opacity(player.isOpen ? 1 : 0)
                .animation(.easeInOut(duration: 0.25), value: player.isOpen)
                .allowsHitTesting(false)

            TabView(selection: $selection) {
                TrainView()
                    .tabItem {
                        Label("训练", systemImage: "figure.strengthtraining.traditional")
                    }
                    .tag(0)

                MusicView()
                    .tabItem {
                        Label("音乐", systemImage: "music.note")
                    }
                    .tag(1)

                GrowthView()
                    .tabItem {
                        Label("成长", systemImage: "chart.xyaxis.line")
                    }
                    .tag(2)

                ProfileView()
                    .tabItem {
                        Label("我的", systemImage: "person.crop.circle")
                    }
                    .tag(3)
            }
            .tint(.accentColor)
            // ★ 4.9 修「一进 App 上下两条大黑边」的真正原因：
            // 以前这里无条件套着 `.clipShape(RoundedRectangle(...))`。TabView 是按安全区
            // 布局的，一旦套上裁剪，即使圆角是 0，也会沿「安全区矩形」硬切一刀 ——
            // 页面延伸到状态栏 / 底部横条上的那部分（导航栏底、标签栏底）被切掉，
            // 露出的就是底下那层，于是上下各留一条边。
            // 现在只在播放页打开（卡片堆叠、需要圆角）时才裁，静止状态完全不裁。
            .modifier(CardClip(active: player.isOpen, radius: 34 - 20 * player.pull))
            // 跟着下拉进度实时变化：拉得越多，下层越接近原始大小、越亮 —— 就是「渐渐回到上一级」
            .scaleEffect(player.isOpen ? 0.94 + 0.06 * player.pull : 1, anchor: .center)
            .overlay {
                Color.black
                    .opacity(player.isOpen ? 0.42 * (1 - Double(player.pull)) : 0)
                    .ignoresSafeArea()
                    .allowsHitTesting(false)
            }
            .allowsHitTesting(!player.isOpen)
            .animation(.spring(response: 0.46, dampingFraction: 0.90), value: player.isOpen)

            if player.isOpen {
                NowPlayingView()
                    .zIndex(10)
                    // 移出时只位移、不淡出：淡出会让播放页在半透明的那几帧里
                    // 透出底下那层（深浅色交接处就会闪一下）。滑出时点一下就能盖住。
                    .transition(.asymmetric(
                        insertion: .move(edge: .bottom).combined(with: .opacity),
                        removal: .move(edge: .bottom)
                    ))
            }
        }
        // 状态栏文字转白：只在播放页已经盖满全屏之后才切（切早了会整块闪）
        .preferredColorScheme(player.forceDark ? ColorScheme.dark : nil)
        .onChange(of: scenePhase) { phase in
            if phase == .background {
                Store.shared.writeNow()
            }
        }
    }
}

struct TrainView: View {
    @ObservedObject var store = Store.shared
    @ObservedObject var rest = RestTimer.shared

    @State private var showPicker = false
    @State private var showFinishAlert = false
    @State private var showDiscardAlert = false

    private let restOptions = [60, 90, 120, 180]

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 16) {
                    exerciseCard

                    if rest.isRunning {
                        restCard
                    }

                    HStack(spacing: 12) {
                        StepperField(label: "重量", unit: "kg",
                                     value: formatWeight(store.lastWeight),
                                     onMinus: { store.setWeight(store.lastWeight - 2.5) },
                                     onPlus: { store.setWeight(store.lastWeight + 2.5) })
                        StepperField(label: "次数", unit: "reps",
                                     value: "\(store.lastReps)",
                                     onMinus: { store.setReps(store.lastReps - 1) },
                                     onPlus: { store.setReps(store.lastReps + 1) })
                    }

                    Button(action: completeSet) {
                        Text("完成一组")
                    }
                    .buttonStyle(GlassButtonStyle(prominent: true))

                    setListCard

                    sessionFooter
                }
                .padding(.horizontal, 20)
                .padding(.bottom, 28)
            }
            .background { AppBackground() }
            .navigationTitle("训练")
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button {
                        showPicker = true
                    } label: {
                        Image(systemName: "list.bullet")
                    }
                }
            }
            .sheet(isPresented: $showPicker) {
                ExercisePickerView()
            }
            .onAppear {
                store.ensureSession()
            }
        }
    }

    // MARK: - 当前动作

    private var exerciseCard: some View {
        let ex = store.currentExercise
        let last = store.lastSession(of: ex.name)
        return SectionCard {
            Button {
                showPicker = true
            } label: {
                HStack {
                    VStack(alignment: .leading, spacing: 9) {
                        Text("当前动作")
                            .font(.caption)
                            .foregroundColor(.secondary)
                        Text(ex.name)
                            .font(.title2)
                            .fontWeight(.bold)
                            .lineSpacing(2)
                            .foregroundColor(.primary)
                        if let last = last {
                            Text("上次 \(last.sets.count) 组 · 最重 \(formatWeight(last.maxWeight)) kg · 1RM \(String(format: "%.1f", last.bestE1RM)) kg")
                                .font(.caption)
                                .lineSpacing(2)
                                .foregroundColor(.secondary)
                                .fixedSize(horizontal: false, vertical: true)
                        } else {
                            Text("还没记录过，先找能标准做 8-12 次的重量")
                                .font(.caption)
                                .lineSpacing(2)
                                .foregroundColor(.secondary)
                        }
                    }
                    Spacer()
                    Image(systemName: "chevron.right")
                        .foregroundColor(.secondary)
                }
            }
            .buttonStyle(.plain)

            if (store.current?.exercises.count ?? 0) > 1 {
                ScrollView(.horizontal, showsIndicators: false) {
                    HStack(spacing: 8) {
                        ForEach(Array((store.current?.exercises ?? []).enumerated()), id: \.element.id) { i, e in
                            Button {
                                store.switchExercise(index: i)
                            } label: {
                                Text(e.name)
                                    .font(.caption)
                                    .padding(.horizontal, 12)
                                    .padding(.vertical, 7)
                                    .background(i == (store.current?.index ?? 0)
                                                ? Color.accentColor.opacity(0.18)
                                                : Color(UIColor.tertiarySystemFill))
                                    .foregroundColor(i == (store.current?.index ?? 0) ? .accentColor : .secondary)
                                    .cornerRadius(20)
                            }
                            .buttonStyle(.plain)
                        }
                    }
                }
            }
        }
    }

    // MARK: - 休息

    private var restCard: some View {
        SectionCard {
            HStack(spacing: 16) {
                RestRing(progress: rest.progress, text: rest.remainingText)
                    .scaleEffect(0.62)
                    .frame(width: 118, height: 118)

                VStack(alignment: .leading, spacing: 12) {
                    HStack(spacing: 6) {
                        ForEach(restOptions, id: \.self) { s in
                            Button {
                                rest.start(seconds: s)
                                store.restSeconds = s
                            } label: {
                                Text("\(s)")
                                    .font(.system(size: 12, weight: s == rest.total ? .semibold : .regular))
                                    .foregroundColor(s == rest.total ? .white : .secondary)
                                    .frame(width: 38, height: 28)
                                    .background(
                                        Capsule().fill(s == rest.total
                                                       ? Color.accentColor
                                                       : Color.primary.opacity(0.06))
                                    )
                            }
                            .buttonStyle(ScaleButtonStyle())
                        }
                    }

                    HStack(spacing: 8) {
                        Button {
                            rest.add(seconds: 30)
                        } label: {
                            Text("+30 秒")
                                .font(.system(size: 12, weight: .medium))
                                .padding(.horizontal, 12)
                                .padding(.vertical, 7)
                                .background(Capsule().fill(Color.primary.opacity(0.06)))
                        }
                        .buttonStyle(ScaleButtonStyle())

                        Button {
                            rest.skip()
                        } label: {
                            Text("跳过")
                                .font(.system(size: 12, weight: .medium))
                                .foregroundColor(.secondary)
                                .padding(.horizontal, 12)
                                .padding(.vertical, 7)
                                .background(Capsule().fill(Color.primary.opacity(0.06)))
                        }
                        .buttonStyle(ScaleButtonStyle())
                    }

                    Text("锁屏后到点由系统通知提醒")
                        .font(.system(size: 10))
                        .lineSpacing(2)
                        .foregroundColor(.secondary)
                }
            }
        }
    }

    // MARK: - 组列表

    private var setListCard: some View {
        let ex = store.currentExercise
        return SectionCard(title: "本次已完成 \(ex.sets.count) 组") {
            if ex.sets.isEmpty {
                Text("还没有记录。调好重量和次数，做完一组点上面的按钮。")
                    .font(.subheadline)
                    .lineSpacing(3)
                    .foregroundColor(.secondary)
            } else {
                VStack(spacing: 0) {
                    ForEach(Array(ex.sets.enumerated()), id: \.element.id) { i, s in
                        HStack {
                            Text("\(i + 1)")
                                .font(.caption)
                                .foregroundColor(.secondary)
                                .frame(width: 22, alignment: .leading)
                            Text("\(formatWeight(s.w)) kg × \(s.r)")
                                .font(.system(.body, design: .rounded))
                                .fontWeight(.medium)
                            Spacer()
                            Text("1RM \(String(format: "%.1f", s.e1rm))")
                                .font(.caption)
                                .foregroundColor(.secondary)
                            Image(systemName: "checkmark")
                                .font(.caption)
                                .foregroundColor(.accentColor)
                        }
                        .padding(.vertical, 13)
                        if i < ex.sets.count - 1 { Divider() }
                    }
                }
                if let last = ex.sets.last {
                    Button {
                        store.deleteSet(last.id)
                    } label: {
                        Label("撤销上一组", systemImage: "arrow.uturn.backward")
                            .font(.caption)
                    }
                    .padding(.top, 8)
                }
            }
        }
    }

    // MARK: - 底部操作

    private var sessionFooter: some View {
        VStack(spacing: 10) {
            Button {
                showFinishAlert = true
            } label: {
                Text("结束训练并保存")
            }
            .buttonStyle(GlassButtonStyle())

            if store.current != nil {
                Button(role: .destructive) {
                    showDiscardAlert = true
                } label: {
                    Text("放弃本次训练")
                        .font(.system(size: 12))
                        .foregroundColor(.secondary)
                }
            }
        }
        .alert("结束本次训练？", isPresented: $showFinishAlert) {
            Button("保存", role: .none) {
                rest.skip()
                store.finishSession()
            }
            Button("取消", role: .cancel) {}
        } message: {
            Text("已完成的组会写入历史，用于成长页统计。")
        }
        .alert("放弃本次训练？", isPresented: $showDiscardAlert) {
            Button("放弃", role: .destructive) {
                rest.skip()
                store.discardSession()
            }
            Button("取消", role: .cancel) {}
        } message: {
            Text("本次记录的所有组都会被删除，不可恢复。")
        }
    }

    // MARK: - 动作

    private func completeSet() {
        _ = store.ensureSession()
        store.logSet(weight: store.lastWeight, reps: store.lastReps)
        UIImpactFeedbackGenerator(style: .medium).impactOccurred()
        rest.start(seconds: store.restSeconds)
    }

    private func formatWeight(_ w: Double) -> String {
        w.truncatingRemainder(dividingBy: 1) == 0 ? String(format: "%.0f", w) : String(format: "%.2f", w)
    }
}

// MARK: - 动作选择

struct ExercisePickerView: View {
    @Environment(\.dismiss) private var dismiss
    @ObservedObject var store = Store.shared

    @State private var search = ""
    @State private var group: String? = nil
    @State private var detail: Move?
    @State private var customName = ""

    private var filtered: [Move] {
        store.library.moves.filter { m in
            let matchGroup = group == nil || m.g == group
            let q = search.trimmingCharacters(in: .whitespaces)
            let matchSearch = q.isEmpty
                || m.n.localizedCaseInsensitiveContains(q)
                || (m.en ?? "").localizedCaseInsensitiveContains(q)
                || (m.mus ?? "").localizedCaseInsensitiveContains(q)
                || (m.eq ?? "").localizedCaseInsensitiveContains(q)
            return matchGroup && matchSearch
        }
    }

    var body: some View {
        NavigationStack {
            List {
                Section {
                    ScrollView(.horizontal, showsIndicators: false) {
                        HStack(spacing: 8) {
                            filterChip(title: "全部", selected: group == nil) { group = nil }
                            ForEach(store.library.groups, id: \.self) { g in
                                filterChip(title: g, selected: group == g) { group = g }
                            }
                        }
                        .padding(.vertical, 4)
                    }
                }
                .listRowInsets(EdgeInsets(top: 4, leading: 16, bottom: 4, trailing: 16))

                Section(header: Text("自定义动作")) {
                    HStack {
                        TextField("不在列表里？输入名字", text: $customName)
                        Button("添加") {
                            let n = customName.trimmingCharacters(in: .whitespaces)
                            guard !n.isEmpty else { return }
                            store.addExercise(name: n)
                            customName = ""
                            dismiss()
                        }
                        .disabled(customName.trimmingCharacters(in: .whitespaces).isEmpty)
                    }
                }

                Section(header: Text("共 \(filtered.count) 个动作")) {
                    ForEach(filtered) { m in
                        HStack {
                            Button {
                                store.addExercise(name: m.n)
                                dismiss()
                            } label: {
                                VStack(alignment: .leading, spacing: 4) {
                                    Text(m.n).foregroundColor(.primary)
                                    Text("\(m.g) · \(m.eq ?? "-") · \(m.p ?? "-")")
                                        .font(.caption)
                                        .foregroundColor(.secondary)
                                }
                            }
                            .buttonStyle(.plain)
                            Spacer()
                            Button {
                                detail = m
                            } label: {
                                Image(systemName: "info.circle")
                                    .foregroundColor(.accentColor)
                            }
                            .buttonStyle(.plain)
                        }
                    }
                }
            }
            .listStyle(.insetGrouped)
            .searchable(text: $search, placement: .navigationBarDrawer(displayMode: .always), prompt: "搜索动作 / 肌肉 / 器械")
            .navigationTitle("选择动作")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button("完成") { dismiss() }
                }
            }
            .sheet(item: $detail) { m in
                MoveDetailView(move: m)
            }
        }
    }

    private func filterChip(title: String, selected: Bool, action: @escaping () -> Void) -> some View {
        Button(action: action) {
            Text(title)
                .font(.subheadline)
                .padding(.horizontal, 14)
                .padding(.vertical, 7)
                .background(selected ? Color.accentColor : Color(UIColor.tertiarySystemFill))
                .foregroundColor(selected ? .white : .primary)
                .cornerRadius(18)
        }
        .buttonStyle(.plain)
    }
}

// MARK: - 动作详情

struct MoveDetailView: View {
    let move: Move
    @Environment(\.dismiss) private var dismiss
    @ObservedObject var store = Store.shared

    var body: some View {
        NavigationStack {
            List {
                Section {
                    VStack(alignment: .leading, spacing: 6) {
                        Text(move.n).font(.title2).fontWeight(.bold)
                        if let en = move.en {
                            Text(en).font(.subheadline).foregroundColor(.secondary)
                        }
                        Text("\(move.g) · \(move.eq ?? "-") · \(move.p ?? "-")")
                            .font(.caption)
                            .foregroundColor(.secondary)
                        if let mus = move.mus {
                            Text(mus).font(.subheadline)
                        }
                    }
                    .padding(.vertical, 4)
                }
                if let cue = move.cue, !cue.isEmpty {
                    Section(header: Text("动作要领")) {
                        ForEach(cue, id: \.self) { Text($0).font(.subheadline) }
                    }
                }
                if let err = move.err, !err.isEmpty {
                    Section(header: Text("常见错误")) {
                        ForEach(err, id: \.self) {
                            Label($0, systemImage: "exclamationmark.triangle")
                                .font(.subheadline)
                                .foregroundColor(.orange)
                        }
                    }
                }
                if let alt = move.alt, !alt.isEmpty {
                    Section(header: Text("替代动作")) {
                        ForEach(alt, id: \.self) { Text($0).font(.subheadline) }
                    }
                }
            }
            .navigationTitle("动作详情")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .navigationBarLeading) {
                    Button("关闭") { dismiss() }
                }
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button("开始练") {
                        store.addExercise(name: move.n)
                        dismiss()
                    }
                }
            }
        }
    }
}
